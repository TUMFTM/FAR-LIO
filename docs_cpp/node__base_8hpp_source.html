85 void init_map()
86 {
87 // Stop the timer if the map is already initialized.
88 if (this->init_status_ != types::InitStatus::WAITING_FOR_MAP) {
89 if (this->init_map_timer_) this->init_map_timer_->cancel();
90 return;
91 }
92 if (!this->map_client_) return;
93 if (!this->map_client_->service_is_ready()) {
94 RCLCPP_WARN(this->get_logger(), "[%s]: Map service not available!", this->get_name());
95 return;
96 }
97 // clang-format off
98 RCLCPP_INFO(this->get_logger(), "[%s]: Requesting initial map from %s.", this->get_name(), this->map_client_->get_service_name()); // NOLINT
99 // clang-format on
100 auto request = std::make_shared<tum_map_msgs::srv::GetPointCloudMap::Request>();
101 std_msgs::msg::Header header;
102 header.stamp = this->get_clock()->now();
103 header.frame_id = this->config_.odom_frame;
104 request->header = header;
105 request->voxel_size = static_cast<float>(this->pipeline_->get_map_resolution());
106 std::function<void(rclcpp::Client<tum_map_msgs::srv::GetPointCloudMap>::SharedFuture)> cb =
107 std::bind(&NodeBase::map_callback, this, std::placeholders::_1, true);
108 this->map_client_->async_send_request(request, cb);
109 // Send a status to the dashboard to indicate that we are waiting for the map
110 this->monitor_->set_message("Waiting for map");
111 this->monitor_->set_error_lvl("map", tam::types::ErrorLvl::WARN);
112 this->monitor_->update();
113 }

169 void update_ready_callback(const std_msgs::msg::Header::SharedPtr msg)
170 {
171 if (this->init_status_ != types::InitStatus::READY) {
172 return;
173 }
174 // Check if an update is already in progress
175 bool expected = false;
176 // clang-format off
177 if (!this->update_in_progress_.compare_exchange_strong(expected, true)) {
178 RCLCPP_WARN(this->get_logger(), "[%s]: Received map update while still in progress - skipping", this->get_name()); // NOLINT
179 return;
180 }
181 this->debug_.map_update = true;
182 // clang-format on
183 // Request map update
184 // Check if we can get the map from the service
185 if (this->update_map_client_ && this->update_map_client_->service_is_ready()) {
186 // clang-format off
187 RCLCPP_INFO(this->get_logger(), "[%s]: Requesting map update from %s.", this->get_name(), this->update_map_client_->get_service_name()); // NOLINT
188 auto request = std::make_shared<tum_map_msgs::srv::GetPointCloudMap::Request>();
189 request->header = *msg;
190 request->voxel_size = static_cast<float>(this->pipeline_->get_map_resolution());
191 std::function<void(rclcpp::Client<tum_map_msgs::srv::GetPointCloudMap>::SharedFuture)> cb =
192 std::bind(&NodeBase::map_callback, this, std::placeholders::_1, false);
193 this->update_map_client_->async_send_request(request, cb);
194 // clang-format on
195 }

205 void map_callback(
206 const rclcpp::Client<tum_map_msgs::srv::GetPointCloudMap>::SharedFuture future, const bool initial_map)
207 {
208 auto response = future.get();
209 const std::string service_name =
210 initial_map ? this->map_client_.get()->get_service_name() : this->update_map_client_.get()->get_service_name();
211 if (!response->success) {
212 // clang-format off
213 RCLCPP_ERROR(this->get_logger(), "[%s]: Map service %s returned failure - skipping", this->get_name(), service_name.c_str()); // NOLINT
214 // clang-format on
215 if (!initial_map) this->update_in_progress_.store(false);
216 return;
217 }
218
219 // Helper lambda to load the map and print the time taken for loading
220 auto load = [&](const tum_map_msgs::srv::GetPointCloudMap::Response::SharedPtr response,
221 const std::string& service_name) {
222 const auto start = std::chrono::high_resolution_clock::now();
223 const sensor_msgs::msg::PointCloud2 cloud = tam::ros::read_pointcloud_file(response->map.pointcloud);
224 tam::ros::remove_pointcloud_file(response->map.pointcloud);
225 // clang-format off
226 const auto time_ms = std::chrono::duration<double, std::milli>(std::chrono::high_resolution_clock::now() - start).count(); // NOLINT
227 RCLCPP_INFO(this->get_logger(), "[%s]: Received map with %u points from %s (%.2f ms)", this->get_name(), cloud.width * cloud.height, service_name.c_str(), time_ms); // NOLINT
228 // clang-format on
229 return utils::cloud2eigen<types::Point_XYZ>(cloud, Eigen::Isometry3f::Identity());
230 };
231
232 if (initial_map) {
233 // Just add the points to the pipeline if its the initial load
234 auto points = load(response, service_name);
235 this->pipeline_->add_points(points);
236 // Publish the map if in debug mode
237 if (this->pipeline_->get_config().debug_mode && this->map_publisher_) {
238 std_msgs::msg::Header header;
239 header.stamp = this->get_clock()->now();
240 header.frame_id = this->config_.odom_frame;
241 this->map_publisher_->publish(utils::eigen2cloud<TConfig>(this->pipeline_->get_map(), header));
242 }
243 // Set the status and cancel the init-map retry timer
244 this->init_status_ = types::InitStatus::WAITING_FOR_EKF;
245 this->monitor_->set_error_lvl("map", tam::types::ErrorLvl::OK);
246 if (this->init_map_timer_) this->init_map_timer_->cancel();
247 } else {
248 // if its an async update, start a new thread to load the map and request the pipeline to
249 // switch the map atomically
250 std::thread([this, response, service_name, load]() {
251 auto points = load(response, service_name);
252 {
253 std::lock_guard<std::mutex> lock(this->pipeline_->get_map_mutex());
254 this->pipeline_->add_points(points, TConfig::MAX_POINTS_PER_VOXEL, 1, TConfig::NUM_NEIGHBORS, false);
255 this->pipeline_->request_map_switch();
256 }
257 // clang-format off
258 RCLCPP_INFO(this->get_logger(), "[%s]: Asynchronous map update applied (%lu points)", this->get_name(), points.size()); // NOLINT
259 // clang-format on
260 this->update_in_progress_.store(false);
261 // Write the debug signal
262 // CAUTION: Set from the worker thread, but should be fine as only read from the main thread
263 // and written by at most one worker thread at the same time
264 this->debug_.map_update = false;
265 }).detach();
266 }
267 }

278 void set_undistortion_poses(const std::uint64_t& stamp)
279 {
280 if (!this->pipeline_->get_config().undistort) {
281 return;
282 }
283 constexpr std::uint64_t UNDISTORTION_INTERVAL_NS = 10'000'000ULL; // 10ms
284 const std::uint64_t frame_ns = stamp;
285 // Round down to the 10ms grid so samples align across frames
286 const std::uint64_t latest_interval_ns = frame_ns - frame_ns % UNDISTORTION_INTERVAL_NS;
287 std::string tf_error;
288 std::uint64_t newest_set_stamp = this->distortion_pose_stamp_;
289 bool gap = false;
290 for (int i = 19; i >= 0; --i) {
291 // clang-format off
292 const std::uint64_t sample_ns = latest_interval_ns - static_cast<std::uint64_t>(i) * UNDISTORTION_INTERVAL_NS; // NOLINT
293 // Skip samples already set in a previous callback
294 if (sample_ns <= this->distortion_pose_stamp_) {
295 continue;
296 }
297 const rclcpp::Time sample_time(sample_ns);
298 if (this->tf_buffer_->canTransform(this->config_.odom_frame, this->config_.child_frame, sample_time, tf2::durationFromSec(0.0), &tf_error)) { // NOLINT
299 this->pipeline_->set_pose_undistortion(utils::transform2pose(this->tf_buffer_->lookupTransform(this->config_.odom_frame, this->config_.child_frame, sample_time))); // NOLINT
300 // Only advance while the fed samples are contiguous (no earlier lookup has failed)
301 if (!gap) {
302 newest_set_stamp = sample_ns;
303 }
304 } else {
305 gap = true;
306 }
307 // clang-format on
308 }
309 this->distortion_pose_stamp_ = newest_set_stamp;
310 }

316 void init_node_common()
317 {
318 // Construct from param manager and logger
319 this->pmg_ = std::make_shared<tam::pmg::ParamReferenceManager>();
320 this->logger_ = std::make_shared<tam::tsl::ReferenceLogger>();
321
322 // Declare node level params and debug values
323 this->set_config(this->pmg_.get());
324 this->set_logging(this->logger_.get());
325
326 // Inizialize odometry pipeline to declare all params
327 this->pipeline_ = OdometryPipeline<TConfig>::from_config(this->pmg_.get(), this->logger_.get());
328
329 // Connect your param manager to the callback
330 // Important: don't forget to store your callback handle here, otherwise it will get
331 // deallocated
332 auto callback_handle = tam::pmg::connect_param_manager_to_ros_cb(this, pmg_);
333 // Declare all paramters from the param manager
334 tam::pmg::declare_ros_params_from_param_manager(this, pmg_.get());
335
336 // Now that the param overrides are active, init necesarry modules
337 // Initialize pipeline
338 this->pipeline_->init();
339
340 // Initialize monitoring of watchdog
341 this->monitor_ = std::make_unique<tam::core::NodeMonitor>(this, this->config_.output_odom);
342 this->topic_watchdog_ = std::make_unique<tam::core::TopicWatchdog>(this);
343
344 // Initialize QoS
345 rclcpp::QoS qos_lidar = tam::ros::get_qos(tam::ros::TopicType::PERCEPTION_SENSOR_DATA);
346 rclcpp::QoS qos_odom = tam::ros::get_qos(tam::ros::TopicType::DEFAULT);
347
348 // Initialize tf listener
349 tf_buffer_ = std::make_unique<tf2_ros::Buffer>(get_clock());
350 tf_listener_ = std::make_unique<tf2_ros::TransformListener>(*tf_buffer_);
351
352 // Initialize logger
353 tsl_publisher_ = std::make_unique<tam::tsl::TSLPublisher>(this, this->logger_);
354
355 // Load static map either from a local .pcd file or from the map loader service
356 if (!this->pipeline_->get_config().update_map) {
357 const std::string& input_map = this->config_.input_map;
358 if (input_map.ends_with(".pcd")) {
359 this->pipeline_->add_points(utils::load_static_map<tam::core::state::types::Point_XYZ>(input_map));
360 this->init_status_ = types::InitStatus::WAITING_FOR_EKF;
361 } else {
362 // Create client to request the map from service (within a separate timer)
363 this->map_client_ = this->create_client<tum_map_msgs::srv::GetPointCloudMap>(input_map);
364 this->init_map_timer_ =
365 tam::create_timer(this, std::chrono::milliseconds(5000), std::bind(&NodeBase::init_map, this));
366 }
367 } else {
368 // If we don't need a map at initialization, we can directly wait for the EKF handshake
369 this->init_status_ = types::InitStatus::WAITING_FOR_EKF;
370 }
371
372 // Intialize subscribers and callbacks
373 // Subscribe tof input pointcloud with optional diagnostics
374 if (this->pipeline_->get_diagnostic_config().check_input_status) {
375 // Both subscriptions need to have the same QoS settings
376 auto sub_pc = topic_watchdog_->add_synced_subscription<sensor_msgs::msg::PointCloud2>(
377 this->config_.input_pointcloud, qos_lidar);
378 auto sub_pc_diag = topic_watchdog_->add_synced_subscription<diagnostic_msgs::msg::DiagnosticArray>(
379 this->config_.input_pointcloud_status, qos_lidar);
380 topic_watchdog_->register_synced_callback(sub_pc, sub_pc_diag,
381 std::bind(&NodeBase::cloud_callback_synced, this, std::placeholders::_1, std::placeholders::_2),
382 std::bind(&NodeBase::timeout_callback_pointcloud, this, std::placeholders::_1, std::placeholders::_2),
383 std::chrono::milliseconds(static_cast<int64_t>(500.0)));
384 } else {
385 // just subscribe to the pointcloud topic without diagnostics using the downstream
386 // callback
387 topic_watchdog_->add_subscription<sensor_msgs::msg::PointCloud2>(this->config_.input_pointcloud, qos_lidar,
388 std::bind(&NodeBase::cloud_callback, this, std::placeholders::_1),
389 std::bind(&NodeBase::timeout_callback_pointcloud, this, std::placeholders::_1, std::placeholders::_2),
390 std::chrono::milliseconds(static_cast<int64_t>(500.0)));
391 }
392
393 // Map Update service and subscriber for update trigger
394 if (!this->config_.update_map_srv.empty()) {
395 this->update_map_client_ = this->create_client<tum_map_msgs::srv::GetPointCloudMap>(this->config_.update_map_srv);
396 const std::string update_ready_topic = this->config_.update_map_srv + "_ready";
397 this->update_ready_sub_ = this->create_subscription<std_msgs::msg::Header>(
398 update_ready_topic, qos_odom, std::bind(&NodeBase::update_ready_callback, this, std::placeholders::_1));
399 }
400
401 // Initialize publishers
402 this->odom_publisher_ = this->create_publisher<nav_msgs::msg::Odometry>(this->config_.output_odom, qos_odom);
403
404 // Debug mode publishers
405 if (this->pipeline_->get_config().debug_mode) {
406 this->frame_publisher_ =
407 this->create_publisher<sensor_msgs::msg::PointCloud2>("/core/state/frame_registration", qos_lidar);
408 if (!this->pipeline_->get_config().update_map) {
409 // If static map is used, only publish once now transient local
410 this->map_publisher_ =
411 this->create_publisher<sensor_msgs::msg::PointCloud2>("/core/state/map", tam::ros::get_qos(tam::ros::EVENT));
412 if (!this->map_client_) {
413 // Map already available from file, publish it now
414 std_msgs::msg::Header header;
415 header.stamp = this->get_clock()->now();
416 header.frame_id = this->config_.odom_frame;
417 this->map_publisher_->publish(utils::eigen2cloud<TConfig>(this->pipeline_->get_map(), header));
418 }
419 } else {
420 // If map is updated, publish with QoS
421 this->map_publisher_ = this->create_publisher<sensor_msgs::msg::PointCloud2>("/core/state/map", qos_lidar);
422 }
423 }
424 // Set static transform for pointcloud
425 this->pc_transform_ = tam::ros::transform2eigen(
426 tam::ros::wait_static_transform(this, tf_buffer_.get(), this->config_.child_frame, this->config_.cloud_frame));
427
428 // Initialize timer
429 this->timer_ = tam::create_timer(this, 100ms, std::bind(&NodeBase::timer_callback, this));
430 }
