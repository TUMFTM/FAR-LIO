97 std::array<float, 36> matrix2array(Eigen::Matrix<float, 6, 6>& cov_matrix, const std::array<float, 6>& min_cov) const
98 {
99 // Initialize array
100 std::array<float, 36> cov_array = {};
101 // Convert matrix to array
102 for (int i = 0; i < 6; ++i) {
103 for (int j = 0; j < 6; ++j) {
104 if (i == j) {
105 cov_array[6 * i + j] = std::max(cov_matrix(i, j), min_cov[i]);
106 } else {
107 cov_array[6 * i + j] = cov_matrix(i, j);
108 }
109 }
110 }
111 return cov_array;
112 }

119 Eigen::Matrix<float, 6, 6> cramer_rao_bound(
120 const std::vector<types::Correspondence<TConfig>>& correspondences, const Sophus::SE3f& pose)
121 requires types::HASNORMALCOV<TConfig> && types::HASSEG<TConfig>
122 {
123 // Initialize variables
124 Eigen::Vector3f frame_p = Eigen::Vector3f::Zero();
125 Eigen::Vector3f map_p = Eigen::Vector3f::Zero();
126 Eigen::Vector3f map_p_normal = Eigen::Vector3f::Zero();
127 std::uint32_t segment_id = 0;
128
129 Eigen::Vector<float, 6> jacobian_i = Eigen::Vector<float, 6>::Zero();
130 Eigen::Matrix<float, 6, 6> JTJ = Eigen::Matrix<float, 6, 6>::Zero();
131 Eigen::Matrix<float, 6, 6> jacobian_i_jacobian_i_T = Eigen::Matrix<float, 6, 6>::Zero();
132 // Translation
133 Eigen::Vector3f pos = pose.translation();
134 // Loop over correspondences
135 for (size_t i = 0; i < correspondences.size(); ++i) {
136 // Capture correspondence components
137 frame_p = correspondences.at(i).frame.pos - pos;
138 map_p = correspondences.at(i).map.pos - pos;
139 map_p_normal = correspondences.at(i).map.normal;
140 segment_id = correspondences.at(i).map.seg;
141 // Skip if no segment
142 if (map_p_normal.hasNaN() || segment_id == 0) {
143 continue;
144 }
145 jacobian_i.segment(0, 3) = -(frame_p.cross(map_p_normal).transpose());
146 jacobian_i.segment(3, 3) = -map_p_normal;
147 jacobian_i_jacobian_i_T = jacobian_i * jacobian_i.transpose();
148 JTJ += jacobian_i_jacobian_i_T;
149 }
150 return JTJ.inverse();
151 }
