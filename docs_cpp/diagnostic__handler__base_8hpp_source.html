60 types::DiagnosticStatus get_diagnostic_status(const types::PoseStamped& pose, const types::TangentStamped& tangent,
61 const types::PoseStamped& init_guess, const types::RegistrationStatus& reg_status)
62 {
63 types::DiagnosticStatus diag_status;
64 // Check registration status
65 // Needs to be called first, because it sets the level of the diagnostic status
66 // Cannot be deactivated
67 this->check_registration(reg_status, diag_status);
68 // Check input status
69 this->check_input_status(diag_status);
70 // Check pose validity
71 this->check_pose(pose, tangent, init_guess, diag_status);
72 // Update previous pose and initial guess
73 if (!this->init_previous_) {
74 this->init_previous_ = true;
75 }
76 this->previous_init_guess_ = init_guess.pose;
77 this->previous_pose_ = pose.pose;
78 // Set debug values
79 this->debug_.overall_status = static_cast<std::int64_t>(diag_status.level);
80 return diag_status;
81 }

149 void check_registration(const types::RegistrationStatus& reg_status, types::DiagnosticStatus& diag_status)
150 {
151 // Set diagnostic status according to convergence of san registration
152 if (!reg_status.converged) {
153 diag_status.level = types::DiagnosticLevel::ERROR;
154 diag_status.message += "not converged";
155 } else if (reg_status.converged && reg_status.duration > this->config_.frame_outdated) {
156 diag_status.level = types::DiagnosticLevel::WARN;
157 diag_status.message += "outdated";
158 } else {
159 diag_status.level = types::DiagnosticLevel::OK;
160 diag_status.message += "converged";
161 }
162 // Update debug info
163 this->debug_.registration.status = static_cast<std::int64_t>(diag_status.level);
164 // Set key value pairs
165 diag_status.key_values["registration_time"] = reg_status.duration;
166 diag_status.key_values["iterations"] = static_cast<double>(reg_status.num_iter);
167 }

173 void check_input_status(types::DiagnosticStatus& diag_status)
174 {
175 if (!this->config_.check_input_status) {
176 return;
177 }
178 // Check if input status is ok
179 if (this->input_status_.level == types::DiagnosticLevel::OK) {
180 // No need to rise the level, because it is already ok
181 diag_status.message += " | cloud ok";
182 this->debug_.input.status = static_cast<std::int64_t>(types::DiagnosticLevel::OK);
183 } else if (this->input_status_.level == types::DiagnosticLevel::WARN) {
184 // if (diag_status.level < 1) diag_status.level = 1;
185 diag_status.message += " | cloud warning";
186 this->debug_.input.status = static_cast<std::int64_t>(types::DiagnosticLevel::WARN);
187 } else if (this->input_status_.level == types::DiagnosticLevel::ERROR) {
188 diag_status.level = types::DiagnosticLevel::ERROR;
189 diag_status.message += " | cloud error";
190 this->debug_.input.status = static_cast<std::int64_t>(types::DiagnosticLevel::ERROR);
191 } else {
192 diag_status.level = types::DiagnosticLevel::ERROR;
193 diag_status.message += " | no cloud status";
194 this->debug_.input.status = static_cast<std::int64_t>(types::DiagnosticLevel::ERROR);
195 }
196 return;
197 }

206 void check_pose(const types::PoseStamped& pose, const types::TangentStamped& tangent,
207 const types::PoseStamped& init_guess, types::DiagnosticStatus& status)
208 {
209 // Check if pose is in front of previous pose
210 if (this->config_.check_forward && !forward_check(pose.pose, init_guess.pose, status)) {
211 return;
212 }
213 // Check if pose is within ellipsis around initial guess
214 if (this->config_.check_ellipsis && !ellipsis_check(pose.pose, init_guess.pose, status)) {
215 return;
216 }
217 // Check if time difference between pose and initial guess is within bounds
218 if (this->config_.check_time_diff && !time_diff_check(pose.stamp, init_guess.stamp, status)) {
219 return;
220 }
221 // Check if velocity is reasonable
222 if (this->config_.check_vel && !vel_check(tangent.tangent, init_guess.pose, status)) {
223 return;
224 }
225 // If all checks passed, set status to valid
226 status.message += " | pose valid";
227 return;
228 }

237 bool forward_check(const Sophus::SE3f& pose, const Sophus::SE3f& init_guess, types::DiagnosticStatus& status)
238 {
239 if (this->init_previous_) {
240 // Check if vehicle has moved
241 if ((this->previous_init_guess_.inverse() * init_guess).translation().norm() >
242 this->config_.min_motion_threshold) {
243 if ((this->previous_pose_.inverse() * pose).translation().x() <= 0.0) {
244 status.level = types::DiagnosticLevel::ERROR;
245 status.message += " | behind previous";
246 this->debug_.pose.status = static_cast<std::int64_t>(types::DiagnosticLevel::ERROR);
247 return false;
248 }
249 }
250 }
251 this->debug_.pose.status = static_cast<std::int64_t>(types::DiagnosticLevel::OK);
252 return true;
253 }

262 bool ellipsis_check(const Sophus::SE3f& pose, const Sophus::SE3f& init_guess, types::DiagnosticStatus& status)
263 {
264 const Sophus::SE3f pose_diff = init_guess.inverse() * pose;
265 this->debug_.pose.diff_initial_guess_s = pose_diff.translation().x();
266 this->debug_.pose.diff_initial_guess_d = pose_diff.translation().y();
267 // Check if point is within ellipsis in s and d
268 if (pow(pose_diff.translation().x() / this->config_.ellipsis_size_s, 2) +
269 pow(pose_diff.translation().y() / this->config_.ellipsis_size_d, 2) <
270 1.0) {
271 this->debug_.pose.status = static_cast<std::int64_t>(types::DiagnosticLevel::OK);
272 return true;
273 } else {
274 status.level = types::DiagnosticLevel::ERROR;
275 status.message += " | too far from guess";
276 this->debug_.pose.status = static_cast<std::int64_t>(types::DiagnosticLevel::ERROR);
277 return false;
278 }
279 }

288 bool time_diff_check(
289 const std::uint64_t pose_stamp, const std::uint64_t init_guess_stamp, types::DiagnosticStatus& status)
290 {
291 std::uint64_t time_diff_ns =
292 (pose_stamp > init_guess_stamp) ? (pose_stamp - init_guess_stamp) : (init_guess_stamp - pose_stamp);
293 this->debug_.pose.diff_time = time_diff_ns * 1.0e-6;
294 if (this->debug_.pose.diff_time < this->config_.max_time_diff) {
295 this->debug_.pose.status = static_cast<std::int64_t>(types::DiagnosticLevel::OK);
296 return true;
297 } else {
298 status.level = types::DiagnosticLevel::ERROR;
299 status.message += " | large time diff";
300 this->debug_.pose.status = static_cast<std::int64_t>(types::DiagnosticLevel::ERROR);
301 return false;
302 }
303 }

312 bool vel_check(const Sophus::SE3f::Tangent& tangent, const Sophus::SE3f& init_guess, types::DiagnosticStatus& status)
313 {
314 // clang-format off
315 if (this->init_previous_) {
316 // Set the status to warn if the vel is disabled
317 if (!this->config_.enable_vel) {
318 // Only set status to warn if its not an error already
319 if (status.level < types::DiagnosticLevel::WARN) status.level = types::DiagnosticLevel::WARN; // NOLINT
320 status.message += " | vel disabled";
321 this->debug_.pose.status = static_cast<std::int64_t>(types::DiagnosticLevel::WARN);
322 return false;
323 }
324 // Check if point is within ellipsis in s and d
325 if (
326 (this->previous_init_guess_.inverse() * init_guess).translation().norm() <
327 this->config_.vel_motion_threshold) {
328 // Only set status to warn if its not an error already
329 if (status.level < types::DiagnosticLevel::WARN) status.level = types::DiagnosticLevel::WARN; // NOLINT
330 status.message += " | below velocity threshold";
331 this->debug_.pose.status = static_cast<std::int64_t>(types::DiagnosticLevel::WARN);
332 return false;
333 } else {
334 // Check if the velocity is valid (i.e. not zeros)
335 if (tangent.head<3>().norm() < this->config_.vel_motion_threshold) {
336 if (status.level < types::DiagnosticLevel::WARN) status.level = types::DiagnosticLevel::WARN; // NOLINT
337 status.message += " | invalid velocity";
338 this->debug_.pose.status = static_cast<std::int64_t>(types::DiagnosticLevel::WARN);
339 return false;
340 }
341 }
342 }
343 this->debug_.pose.status = static_cast<std::int64_t>(types::DiagnosticLevel::OK);
344 return true;
345 // clang-format on
346 }
