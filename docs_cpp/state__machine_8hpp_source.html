82 void update(bool overwrite_state_machine = false)
83 {
84 if (previous_param_state_hash_ != param_manager_->get_state_hash()) {
85 this->declare_and_update_parameters();
86 }
87 // check all valid whether atleast one input source per modalitiy is valid
88 bool valid_loc = false, valid_lin_vel = false, valid_imu = false;
89 bool valid_pos = false, valid_orientation = false, imu_safe_stop = false;
90
91 // check whether there is a valid position input
92 for (uint8_t i = 0; i < TConfig::NUM_POS_MEASUREMENT; ++i)
93 valid_pos |= valid_map_[{tam::types::state::measurements::POS, i}];
94
95 // check whether there is a valid orientation input
96 // dont consider the last two measurements because they are reserved for internal usage
97 for (uint8_t i = 0; i < TConfig::NUM_ORIENTATION_MEASUREMENT - 2; ++i)
98 valid_orientation |= valid_map_[{tam::types::state::measurements::ORIENTATION, i}];
99
100 // combine the valid bits for position and orientation to obtain the a valid localization
101 valid_loc = valid_orientation & valid_pos;
102
103 // check whether there is a valid linear velocity input
104 for (uint8_t i = 0; i < TConfig::NUM_VEL_MEASUREMENT; ++i)
105 valid_lin_vel |= valid_map_[{tam::types::state::measurements::VEL, i}];
106
107 // check whether there is a valid imu input
108 for (uint8_t i = 0; i < (TConfig::NUM_IMU_MEASUREMENT + TConfig::NUM_BACKUP_IMU_MEASUREMENT); ++i)
109 valid_imu |= valid_map_[{tam::types::state::measurements::IMU, i}];
110
111 // perform a safe stop if the P_VDC_MinValidIMUs threshold is voilated or the backup imu mode
112 // is activated (only active if one lin_vel and loc is valid)
113 // clang-format off
114 bool backup_imu_valid = valid_map_[{tam::types::state::measurements::IMU, TConfig::NUM_IMU_MEASUREMENT + TConfig::NUM_BACKUP_IMU_MEASUREMENT - 1}]; // NOLINT
115 if (get_num_valid_imus() < state_machine_params_.min_valid_imus) imu_safe_stop = true;
116 if (get_num_valid_imus() == 0 && backup_imu_active_ && backup_imu_valid) imu_safe_stop = true;
117 // clang-format on
118
119 // check whether more than the state estimation internal linear velocity input is valid
120 // this indicates that a valid vehicle side slip can be estimated
121 bool valid_side_slip = get_num_valid_vels() > 1;
122
123 // overwrite_state_machine allows any state transition
124 // this should only be used to initialize or recover the state machine
125 if (overwrite_state_machine) {
126 if (valid_loc & valid_imu & valid_lin_vel & valid_side_slip & !imu_safe_stop) {
127 state_ = tam::types::ErrorLvl::OK;
128 status_message_ = "OK";
129 } else if (valid_loc & valid_imu & valid_lin_vel & !valid_side_slip & !imu_safe_stop) {
130 state_ = tam::types::ErrorLvl::WARN;
131 status_message_ = "no valid ssa";
132 } else if (((!imu_safe_stop) | (imu_safe_stop & valid_lin_vel & valid_loc)) & valid_imu) {
133 state_ = tam::types::ErrorLvl::ERROR;
134 status_message_ = "no valid:";
135 if (!valid_loc) status_message_ += " loc";
136 if (!valid_lin_vel) status_message_ += " lin_vel";
137 if (imu_safe_stop) status_message_ += " main_imu";
138 } else {
139 state_ = tam::types::ErrorLvl::STALE;
140 status_message_ = "no valid imu";
141 }
142 } else {
143 // set the state estimation status
144 // if the state estimation status is OK or WARN all states can be reached
145 if (state_ == tam::types::ErrorLvl::OK || state_ == tam::types::ErrorLvl::WARN) {
146 if (valid_loc & valid_imu & valid_lin_vel & valid_side_slip & !imu_safe_stop) {
147 state_ = tam::types::ErrorLvl::OK;
148 status_message_ = "OK";
149 } else if (valid_loc & valid_imu & valid_lin_vel & !valid_side_slip & !imu_safe_stop) {
150 state_ = tam::types::ErrorLvl::WARN;
151 status_message_ = "no valid ssa";
152 } else if (((!imu_safe_stop) | (imu_safe_stop & valid_lin_vel & valid_loc)) & valid_imu) {
153 state_ = tam::types::ErrorLvl::ERROR;
154 status_message_ = "no valid:";
155 if (!valid_loc) status_message_ += " loc";
156 if (!valid_lin_vel) status_message_ += " lin_vel";
157 if (imu_safe_stop) status_message_ += " main_imu";
158 } else {
159 state_ = tam::types::ErrorLvl::STALE;
160 status_message_ = "no valid imu";
161 }
162 // if the state estimation is in state Error the Error
163 // is buffered and the state can only get worse
164 } else if (state_ == tam::types::ErrorLvl::ERROR) {
165 if (((!imu_safe_stop) | (imu_safe_stop & valid_lin_vel & valid_loc)) & valid_imu) {
166 state_ = tam::types::ErrorLvl::ERROR;
167 status_message_ = "no valid:";
168 if (!valid_loc) status_message_ += " loc";
169 if (!valid_lin_vel) status_message_ += " lin_vel";
170 if (imu_safe_stop) status_message_ += " main_imu";
171 if (valid_loc && valid_lin_vel && !imu_safe_stop) status_message_ = "error buffered";
172 } else {
173 state_ = tam::types::ErrorLvl::STALE;
174 status_message_ = "no valid imu";
175 }
176 // if the state estimation is already in STALE it can only stay in STALE
177 } else {
178 state_ = tam::types::ErrorLvl::STALE;
179 if (!valid_imu) {
180 status_message_ = "no valid imu";
181 } else {
182 status_message_ = "stale buffered";
183 }
184 }
185 }

197 void set_position_status(const tam::types::ErrorLvl& status, uint8_t pos_num)
198 {
199 // set the position input valid if the sensor status is OK (0)
200 if (pos_num >= TConfig::NUM_POS_MEASUREMENT) return;
201 if (status == tam::types::ErrorLvl::OK || status == tam::types::ErrorLvl::WARN) {
202 valid_map_[{tam::types::state::measurements::POS, pos_num}] = true;

266 void set_imu_status(const tam::types::ErrorLvl& status, uint8_t imu_num)
267 {
268 // set the imu input valid if the sensor status is OK (0)
269 if (imu_num >= TConfig::NUM_IMU_MEASUREMENT + TConfig::NUM_BACKUP_IMU_MEASUREMENT) return;
270 if (status == tam::types::ErrorLvl::OK) {
271 valid_map_[{tam::types::state::measurements::IMU, imu_num}] = true;
272 if (imu_num < TConfig::NUM_IMU_MEASUREMENT) set_backup_imu_inactive();
273 } else {
274 // handle backup imu
275 bool prev_status = valid_map_[{tam::types::state::measurements::IMU, imu_num}];
276 if (imu_num < TConfig::NUM_IMU_MEASUREMENT && get_num_valid_imus() == 1 && prev_status) {
277 set_backup_imu_active();
278 }
279 valid_map_[{tam::types::state::measurements::IMU, imu_num}] = false;
280 }
281
282 // update the state machine
283 update();

399 Eigen::Vector<double, TConfig::MEASUREMENT_VECTOR_SIZE> get_valid_fusion_vec(void)
400 {
401 Eigen::Vector<double, TConfig::MEASUREMENT_VECTOR_SIZE> valid_fusion_vec;
402 valid_fusion_vec.setZero();
403
404 // check all position valid bits
405 // clang-format off
406 for (uint8_t i = 0; i < TConfig::NUM_POS_MEASUREMENT; ++i) {
407 if (valid_map_[{tam::types::state::measurements::POS, i}]) {
408 valid_fusion_vec.segment(i * TConfig::POS_MEASUREMENT_VECTOR_SIZE, TConfig::POS_MEASUREMENT_VECTOR_SIZE).setOnes(); // NOLINT
409 }
410 }
411 // clang-format on
412
413 // check all orientation valid bits
414 // clang-format off
415 for (uint8_t i = 0; i < TConfig::NUM_ORIENTATION_MEASUREMENT; ++i) {
416 if (valid_map_[{tam::types::state::measurements::ORIENTATION, i}]) {
417 valid_fusion_vec.segment(TConfig::MEASUREMENT_VECTOR_OFFSET_ORIENTATION + i * TConfig::ORIENTATION_MEASUREMENT_VECTOR_SIZE, TConfig::ORIENTATION_MEASUREMENT_VECTOR_SIZE).setOnes(); // NOLINT
418 }
419 }
420 // clang-format on
421
422 // check linear velocity valid bits
423 // clang-format off
424 for (uint8_t i = 0; i < TConfig::NUM_VEL_MEASUREMENT; ++i) {
425 if (valid_map_[{tam::types::state::measurements::VEL, i}]) {
426 valid_fusion_vec.segment(TConfig::MEASUREMENT_VECTOR_OFFSET_VEL + i * TConfig::VEL_MEASUREMENT_VECTOR_SIZE, TConfig::VEL_MEASUREMENT_VECTOR_SIZE).setOnes(); // NOLINT
427 }
428 }
429 // clang-format on
430
431 return valid_fusion_vec;
432 };
