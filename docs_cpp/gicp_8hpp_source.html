37template <typename TConfig>
38struct GICPFactor : public FactorBase<TConfig> {
39 types::LinearSystem operator()(types::Correspondence<TConfig>& correspondence) const override
40 {
41 // Compute precision matrix from correspondence covs
42 correspondence.precision = Eigen::Matrix3f::Identity();
43 if (!correspondence.frame.cov.isZero(1.0e-6f) && !correspondence.map.cov.isZero(1.0e-6f)) {
44 const Eigen::Matrix3f mahalanobis = (correspondence.map.cov + correspondence.frame.cov).inverse();
45 // Make sure the covariance is not singular
46 correspondence.precision = (mahalanobis.hasNaN()) ? Eigen::Matrix3f::Zero() : mahalanobis;
47 }
48 const Eigen::Vector3f residual = correspondence.frame.pos - correspondence.map.pos;
49 Eigen::Matrix<float, 3, 6> J_r;
50 J_r.block<3, 3>(0, 0) = Eigen::Matrix3f::Identity();
51 J_r.block<3, 3>(0, 3) = -1.0 * Sophus::SO3f::hat(correspondence.frame.pos);
52 // Compute kernel weight
53 const float w = robust_kernel_weight<TConfig>(residual, this->kernel_scale_);
54 return types::LinearSystem(J_r.transpose() * w * correspondence.precision * J_r, // JTJ
55 J_r.transpose() * w * correspondence.precision * residual); // JTr
56 }
57};

67template <typename TConfig>
68struct GICPError : public ErrorBase<TConfig> {
69 float operator()(const types::Correspondence<TConfig>& correspondence) const override
70 {
71 // Compute residual with double precision
72 const Eigen::Vector3f residual = correspondence.map.pos - this->T_ * correspondence.frame.pos;
73 // Compute error
74 return 0.5f * residual.transpose() * correspondence.precision * residual;
75 }
76};

112 Sophus::SE3f register_frame(const std::vector<types::Point<TConfig>>& frame, const MapHandler<TConfig>* map,
113 const Sophus::SE3f& initial_guess, const float correspondence_threshold, const float kernel_scale) override
114 {
115 if (map->empty()) {
116 this->debug_.conditional["frame_map_time"] = 0.0;
117 return initial_guess;
118 }
119 // Start timer
120 auto start = std::chrono::high_resolution_clock::now();
121
122 // Add points to the local map instance to trigger normal/covariance computation
123 std::vector<types::Point<tam::core::state::types::Point_XYZ>> point_xyz;
124 point_xyz.reserve(frame.size());
125 std::transform(frame.begin(), frame.end(), std::back_inserter(point_xyz), [](const auto& pt) {
126 return tam::core::state::utils::convert_point<TConfig, tam::core::state::types::Point_XYZ>(pt);
127 });
128 this->frame_map_->add_points(point_xyz);
129
130 // Get back points with normals
131 std::vector<types::Point<TConfig>> source = this->frame_map_->get_cloud();
132 utils::transform_points(initial_guess, source);
133
134 // Compute elapsed time
135 // clang-format off
136 const double time = std::chrono::duration_cast<std::chrono::microseconds>(std::chrono::high_resolution_clock::now() - start).count() * 1.0e-3; // NOLINT
137 // clang-format on
138 this->debug_.conditional["frame_map_time"] = time;
139
140 // Call the solver
141 const Sophus::SE3f T =
142 this->solve(source, map, this->gicp_factor_, this->gicp_error_, correspondence_threshold, kernel_scale);
143
144 // Spit the final transformation
145 return T * initial_guess;
146 }
