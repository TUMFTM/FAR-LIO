39// Macro to initialize modules from config
40#define INIT_MODULE(PARAM, TYPE, NAME, MODULE, CONFIG, DEBUG) \
41 if constexpr (TConfig::PARAM == types::TYPE) { \
42 this->NAME = MODULE<TConfig>::from_config(CONFIG, DEBUG); \
43 std::cout << "\033[1;36mInitialized module " << #MODULE << "!\033[0m" << std::endl; \
44 }
45// Macro to copy fields of different point types
46// FLAG is the TConfig switch guarding the field, MEMBER the corresponding types::Point member.
47// Both are needed because they differ (e.g. NORMALS guards `normal`); naming them with a single
48// token silently skips the copy instead of failing to compile.
49#define CONDITIONAL_COPY(FLAG, MEMBER) \
50 if constexpr (TConfigSrc::FLAG && TConfigDst::FLAG) { \
51 dst.MEMBER = src.MEMBER; \
52 }

60template <typename TConfig>
61inline void transform_points(const Sophus::SE3f& T, std::vector<types::Point<TConfig>>& points)
62{
63 std::transform(points.cbegin(), points.cend(), points.begin(), [&](const auto& point) {
64 types::Point<TConfig> pt = point;
65 pt.pos = T * point.pos;
66 // Transform normals and covariances if they exist
67 if constexpr (TConfig::NORMALS) {
68 pt.normal = T.rotationMatrix() * point.normal;
69 }
70 if constexpr (TConfig::COV) {
71 pt.cov = T.rotationMatrix() * point.cov * T.rotationMatrix().transpose();
72 }
73 return pt;
74 });
75}

81template <typename TConfigSrc, typename TConfigDst>
82inline types::Point<TConfigDst> convert_point(const types::Point<TConfigSrc>& src)
83{
84 types::Point<TConfigDst> dst;
85 // Unconditional members of types::Point.
86 dst.pos = src.pos;
87 dst.timestamp = src.timestamp;
88 CONDITIONAL_COPY(SPHERICAL, azimuth);
89 CONDITIONAL_COPY(SPHERICAL, range);
90 CONDITIONAL_COPY(SPHERICAL, elevation);
91 CONDITIONAL_COPY(INTENSITY, intensity);
92 CONDITIONAL_COPY(NORMALS, normal);
93 CONDITIONAL_COPY(COV, cov);
94 CONDITIONAL_COPY(SEG, seg);
95 CONDITIONAL_COPY(VEL, vel);
96 CONDITIONAL_COPY(RCS, rcs);
97 CONDITIONAL_COPY(SNR, snr);
98 CONDITIONAL_COPY(CONFIDENCE, confidence);
99 CONDITIONAL_COPY(VEL_INTERVAL, vel_interval);
100 CONDITIONAL_COPY(SENSOR_ID, sensor_id);
101 return dst;
102}

111template <typename TConfig>
112inline std::vector<types::Point<TConfig>> load_pointcloud_bin(const std::string& path)
113{
114 std::vector<types::Point<TConfig>> point_cloud;
115
116 // Open file
117 FILE* stream = fopen(path.c_str(), "rb");
118 if (!stream) {
119 throw std::logic_error("Failed to open file: " + path);
120 }
121
122 // Determine file size
123 fseek(stream, 0, SEEK_END);
124 size_t file_size = ftell(stream);
125 rewind(stream);
126
127 size_t num_floats = file_size / sizeof(float);
128 if (num_floats % 4 != 0) {
129 fclose(stream);
130 throw std::logic_error("File size does not match expected format: " + path);
131 }
132
133 // Allocate buffer
134 std::unique_ptr<float[]> data(new float[num_floats]);
135 size_t read_count = fread(data.get(), sizeof(float), num_floats, stream);
136 fclose(stream);
137
138 // Check if the correct number of floats was read
139 if (read_count != num_floats) {
140 throw std::runtime_error("Failed to read the expected number of floats from file: " + path);
141 }
142
143 // Parse points
144 for (size_t i = 0; i < num_floats; i += 4) {
145 types::Point<TConfig> pt;
146 pt.pos.x() = data[i];
147 pt.pos.y() = data[i + 1];
148 pt.pos.z() = data[i + 2];
149 point_cloud.push_back(pt);
150 }
151
152 return point_cloud;
153}

162template <typename TConfig>
163inline Sophus::SE3f load_pose(const std::string& path)
164{
165 Sophus::SE3f pose;
166 double x, y, z, qw, qx, qy, qz, x_stddev, y_stddev, z_stddev;
167
168 std::ifstream infile(path);
169 std::string line;
170
171 while (std::getline(infile, line)) {
172 std::istringstream iss(line);
173
174 if (!(iss >> x >> y >> z >> qw >> qx >> qy >> qz >> x_stddev >> y_stddev >> z_stddev)) {
175 std::cerr << "Pose in wrong format!" << std::endl;
176 return pose;
177 }
178 pose = Sophus::SE3f(Sophus::SE3f::QuaternionType(qw, qx, qy, qz), Sophus::SE3f::Point(x, y, z));
179 }
180 return pose;
181}

209template <typename TConfig>
210inline std::tuple<std::vector<rerun::Position3D>, std::vector<rerun::Color>> points2rerun(
211 const std::vector<types::Point<TConfig>>& points, const std::string& color, const Sophus::SE3f& T = Sophus::SE3f())
212{
213 std::vector<rerun::Position3D> positions{};
214 std::vector<rerun::Color> colors{};
215 TUMcolor tum_color(color);
216 positions.reserve(points.size());
217 colors.reserve(points.size());
218 for (const auto& point : points) {
219 Eigen::Vector3f pos = (T * point.pos).template cast<float>();
220 positions.push_back(rerun::Position3D(pos.x(), pos.y(), pos.z()));
221 colors.push_back(rerun::Color(tum_color.r, tum_color.g, tum_color.b, tum_color.a));
222 }
223 return std::make_tuple(positions, colors);
224}

235template <typename TConfig>
236inline std::tuple<std::vector<rerun::Position3D>, std::vector<rerun::Color>, std::vector<rerun::Vector3D>>
237points_normals2rerun(const std::vector<types::Point<TConfig>>& points, const std::string& color)
238 requires types::HASNORMALCOV<TConfig>
239{
240 std::vector<rerun::Position3D> positions{};
241 std::vector<rerun::Vector3D> normals{};
242 std::vector<rerun::Color> colors{};
243 TUMcolor tum_color(color);
244 positions.reserve(points.size());
245 normals.reserve(points.size());
246 colors.reserve(points.size());
247 for (const auto& point : points) {
248 Eigen::Vector3f pos = point.pos.template cast<float>();
249 Eigen::Vector3f normal = point.normal.template cast<float>();
250 positions.push_back(rerun::Position3D(pos.x(), pos.y(), pos.z()));
251 normals.push_back(rerun::Vector3D(normal.x(), normal.y(), normal.z()));
252 colors.push_back(rerun::Color(tum_color.r, tum_color.g, tum_color.b, tum_color.a));
253 }
254 return std::make_tuple(positions, colors, normals);
255}
