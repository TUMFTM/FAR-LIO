39TEST_P(ICPParameterizedTest, ICPRegisterFrameParameterized)
40{
41 const auto& params = GetParam();
42 // Initialize ICP
43 // clang-format off
44 // Construct from param manager and logger
45 tam::pmg::ParamReferenceManager::UniquePtr pmg_ = std::make_unique<tam::pmg::ParamReferenceManager>(); // NOLINT
46 tam::tsl::ReferenceLogger::UniquePtr logger_ = std::make_unique<tam::tsl::ReferenceLogger>(); // NOLINT
47 #ifdef __CUDACC__
48 // Check if CUDA device is available
49 CHECK_CUDA_ERROR();
50 auto map_ = tam::core::state::cuda::VoxelHashMap<tam::core::state::types::CUDA_ICP_EXT>::from_config(pmg_.get(), logger_.get()); // NOLINT
51 auto registration_ = tam::core::state::cuda::ICP<tam::core::state::types::CUDA_ICP_EXT>::from_config(pmg_.get(), logger_.get()); // NOLINT
52 #else
53 auto map_ = tam::core::state::VoxelHashMap<tam::core::state::types::ICP_EXT>::from_config(pmg_.get(), logger_.get()); // NOLINT
54 auto registration_ = tam::core::state::ICP<tam::core::state::types::ICP_EXT>::from_config(pmg_.get(), logger_.get()); // NOLINT
55 #endif
56 // clang-format on
57 tam::pmg::MgmtInterface* pmg_raw = pmg_.get();
58 pmg_raw->set_value("map.frame_map", false);
59 pmg_raw->set_value("map.voxel_size", 1.0);
60 pmg_raw->set_value("map.max_distance", 100.0);
61 pmg_raw->set_value("map.cov_regularization", params.cov_regularization); // not used here
62 pmg_raw->set_value("registration.solver_type", params.solver_type);
63 pmg_raw->set_value("registration.damping_factor", 0.0);
64 pmg_raw->set_value("registration.convergence_criterion", 5.0e-3);
65 map_->init();
66 registration_->init();
67
68 // Define a frame of points
69 std::vector<tam::core::state::types::Point<tam::core::state::types::Point_XYZ>> frame =
70 generate_points<tam::core::state::types::Point_XYZ>(Sophus::SE3f());
71
72 map_->add_points(frame);
73 // clang-format on
74
75 Sophus::SE3f trans =
76 Sophus::SE3f(Sophus::SE3f::QuaternionType(0.50, 0.004, 0.02, 0.86), Sophus::SE3f::Point(3.0, 2.0, 5.0));
77
78 // Transform points
79#ifdef __CUDACC__
80 std::vector<tam::core::state::types::Point<tam::core::state::types::CUDA_ICP_EXT>> transformed_frame_h =
81 generate_points<tam::core::state::types::CUDA_ICP_EXT>(trans);
82 thrust::device_vector<tam::core::state::types::Point<tam::core::state::types::CUDA_ICP_EXT>> transformed_frame(
83 transformed_frame_h.begin(), transformed_frame_h.end());
84#else
85 std::vector<tam::core::state::types::Point<tam::core::state::types::ICP_EXT>> transformed_frame =
86 generate_points<tam::core::state::types::ICP_EXT>(trans);
87#endif
88
89 // Generate initial guess
90 Sophus::SE3f init_guess = trans.inverse();
91 std::default_random_engine generator;
92 std::normal_distribution<float> quaternion_distribution_init(0.05, 0.001); // Mean 0, Stddev 0.01
93 std::normal_distribution<float> translation_distribution_init(1.0, 0.1); // Mean 0.1, Stddev 0.1
94 Sophus::SE3f init_guess_noise = init_guess;
95 init_guess_noise.translation().x() += translation_distribution_init(generator);
96 init_guess_noise.translation().y() += translation_distribution_init(generator);
97 init_guess_noise.translation().z() += translation_distribution_init(generator);
98 // Apply random noise to the initial guess quaternion
99 Sophus::SE3f::QuaternionType init_guess_quaternion = init_guess_noise.unit_quaternion();
100 init_guess_quaternion.w() += quaternion_distribution_init(generator);
101 init_guess_quaternion.x() += quaternion_distribution_init(generator);
102 init_guess_quaternion.y() += quaternion_distribution_init(generator);
103 init_guess_quaternion.z() += quaternion_distribution_init(generator);
104 init_guess_quaternion.normalize(); // Ensure the quaternion remains valid
105 init_guess_noise.setQuaternion(init_guess_quaternion);
106
107 const float sigma = 6.0;
108 const Sophus::SE3f T_icp =
109 registration_->register_frame(transformed_frame, map_.get(), init_guess_noise, 3.0 * sigma, sigma / 3.0);
110
111 std::cout << "Testing with solver_type: " << params.solver_type << std::endl;
112 std::cout << "Iterations: " << registration_->get_debug().num_iter << std::endl;
113
114 EXPECT_NEAR(T_icp.translation().x(), init_guess.translation().x(), 1.0e-1)
115 << "Wrong translation in x-direction for " << params.string();
116 EXPECT_NEAR(T_icp.translation().y(), init_guess.translation().y(), 1.0e-1)
117 << "Wrong translation in y-direction for " << params.string();
118 EXPECT_NEAR(T_icp.translation().z(), init_guess.translation().z(), 1.0e-1)
119 << "Wrong translation in z-direction for " << params.string();
120 EXPECT_NEAR(T_icp.unit_quaternion().w(), init_guess.unit_quaternion().w(), 1.0e-2)
121 << "Wrong quaternion w-component for " << params.string();
122 EXPECT_NEAR(T_icp.unit_quaternion().x(), init_guess.unit_quaternion().x(), 1.0e-2)
123 << "Wrong quaternion x-component for " << params.string();
124 EXPECT_NEAR(T_icp.unit_quaternion().y(), init_guess.unit_quaternion().y(), 1.0e-2)
125 << "Wrong quaternion y-component for " << params.string();
126 EXPECT_NEAR(T_icp.unit_quaternion().z(), init_guess.unit_quaternion().z(), 1.0e-2)
127 << "Wrong quaternion z-component for " << params.string();
128}
