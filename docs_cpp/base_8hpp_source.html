74 void update_measurement_covariance_matrix(const tam::types::state::unordered_identifier_map<bool>& valid_map)
75 {
76 // get the configured Measurement covarinace matrix
77 const Eigen::Map<const Eigen::VectorXd> R_configured(
78 kalman_filter_params_.R_init.data(), kalman_filter_params_.R_init.size());
79
80 // clang-format off
81 for (int i = 0; i < R_adaptive_.size(); ++i) {
82 double covariance_limit = kalman_filter_params_.covariance_adaption_limit;
83 double covariance_decay = kalman_filter_params_.covariance_adaption_decay;
84 if (i >= TConfig::MEASUREMENT_VECTOR_OFFSET_ORIENTATION) {
85 covariance_limit /= 100;
86 covariance_decay /= 5;
87 }
88 // ensure no elements are set to zero
89 if (std::abs(R_adaptive_(i)) < R_configured(i)) R_adaptive_(i) = R_configured(i);
90
91 // decrease the variance that was set to the limit after a change in the valid bits
92 if (R_(i, i) > R_adaptive_[i]) {
93 if (R_(i, i) > covariance_limit && R_adaptive_[i] < covariance_limit) {
94 R_(i, i) = covariance_limit;
95 R_decay_[i] = static_cast<uint16_t>(covariance_limit / covariance_decay);
96 }
97 if (((R_(i, i) < covariance_limit || R_adaptive_[i] < covariance_limit) && R_(i, i) > covariance_decay * 2) && R_decay_[i] > 0) { // NOLINT
98 R_(i, i) -= covariance_decay;
99 R_decay_[i]--;
100 } else {
101 R_(i, i) = R_adaptive_[i];
102 }
103 } else {
104 R_(i, i) = R_adaptive_[i];
105 }
106 }
107
108 // check all position valid bits
109 for (uint8_t i = 0; i < TConfig::NUM_POS_MEASUREMENT; ++i) {
110 const tam::types::state::measurements::identifier key = {tam::types::state::measurements::POS, i};
111 if (valid_map.at(key) != previous_valid_map_.at(key)) {
112 // set all measurement covariances to the limit if a valid bit changes
113 R_(i * TConfig::POS_MEASUREMENT_VECTOR_SIZE + TConfig::MEASUREMENT_POS_X_M, i * TConfig::POS_MEASUREMENT_VECTOR_SIZE + TConfig::MEASUREMENT_POS_X_M) = kalman_filter_params_.covariance_adaption_limit; // NOLINT
114 R_(i * TConfig::POS_MEASUREMENT_VECTOR_SIZE + TConfig::MEASUREMENT_POS_Y_M, i * TConfig::POS_MEASUREMENT_VECTOR_SIZE + TConfig::MEASUREMENT_POS_Y_M) = kalman_filter_params_.covariance_adaption_limit; // NOLINT
115 if constexpr (TConfig::POS_MEASUREMENT_VECTOR_SIZE > 2) {
116 R_(i * TConfig::POS_MEASUREMENT_VECTOR_SIZE + TConfig::MEASUREMENT_POS_Z_M, i * TConfig::POS_MEASUREMENT_VECTOR_SIZE + TConfig::MEASUREMENT_POS_Z_M) = kalman_filter_params_.covariance_adaption_limit; // NOLINT
117 }
118 }
119 }
120
121 // check all orientation valid bits
122 for (uint8_t i = 0; i < TConfig::NUM_ORIENTATION_MEASUREMENT; ++i) {
123 const tam::types::state::measurements::identifier key = {tam::types::state::measurements::ORIENTATION, i};
124 if (valid_map.at(key) != previous_valid_map_.at(key)) {
125 // set all measurement covariances to the limit if a valid bit changes
126 const int orientation_offset = TConfig::MEASUREMENT_VECTOR_OFFSET_ORIENTATION + i * TConfig::ORIENTATION_MEASUREMENT_VECTOR_SIZE; // NOLINT
127 R_(orientation_offset + TConfig::MEASUREMENT_PSI_RAD, orientation_offset + TConfig::MEASUREMENT_PSI_RAD) = kalman_filter_params_.covariance_adaption_limit / 100; // NOLINT
128 if constexpr (TConfig::ORIENTATION_MEASUREMENT_VECTOR_SIZE > 1) {
129 R_(orientation_offset + TConfig::MEASUREMENT_PHI_RAD, orientation_offset + TConfig::MEASUREMENT_PHI_RAD) = kalman_filter_params_.covariance_adaption_limit / 100; // NOLINT
130 R_(orientation_offset + TConfig::MEASUREMENT_THETA_RAD, orientation_offset + TConfig::MEASUREMENT_THETA_RAD) = kalman_filter_params_.covariance_adaption_limit / 100; // NOLINT
131 }
132 }
133 }
134 // clang-format on
135
136 previous_valid_map_ = valid_map;
137 }

142 void set_covariance_matricies(void)
143 {
144 // update parameters
145 if (param_manager_->get_state_hash() != previous_param_state_hash_) {
146 this->declare_and_update_parameters();
147 }
148
149 // set the Measurement Noise Covariance Matrix
150 R_.diagonal() =
151 Eigen::Map<Eigen::VectorXd>(kalman_filter_params_.R_init.data(), kalman_filter_params_.R_init.size());
152 R_adaptive_ = Eigen::Map<Eigen::VectorXd>(kalman_filter_params_.R_init.data(), kalman_filter_params_.R_init.size());
153
154 // set the Process Noise Covariance Matrix
155 Q_.diagonal() = Eigen::Map<Eigen::VectorXd>(kalman_filter_params_.Q.data(), kalman_filter_params_.Q.size());
156
157 // set the outlier bound vector to only load the data from the param manager once
158 outlier_bound_ = Eigen::Map<Eigen::VectorXd>(
159 kalman_filter_params_.outlier_bounds.data(), kalman_filter_params_.outlier_bounds.size());
160 };

168 void update_measurement_matrix(
169 const Eigen::Ref<const Eigen::Vector<double, TConfig::MEASUREMENT_VECTOR_SIZE>>& fusion_vec)
170 {
171 // extraction matrix which only uses the sensors configured and updated
172 // buffer the fusion vector for the update step of the filter
173 fusion_vec_ = fusion_vec;
174
175 // update the measurement matrix such that we only use updated measurements in the
176 // update step of the Kalman filter (diagonal row-scaling of H_full_)
177 H_.noalias() = fusion_vec.asDiagonal() * H_full_;
178 };

206 void set_position_covariance(
207 const Eigen::Ref<const Eigen::Vector<double, TConfig::POS_MEASUREMENT_VECTOR_SIZE>>& covariance, uint8_t pos_num)
208 {
209 // set the variance in the diagonal of the measurement noise covariance matrix for the position
210 // clang-format off
211 if (pos_num >= TConfig::NUM_POS_MEASUREMENT)
212 throw std::invalid_argument(
213 "[StateEstimationCPP]: Position input " + std::to_string(pos_num) + " does not exist");
214 if (covariance.array().isNaN().any()) {
215 // If any element is NaN, set all elements in the specified segment to zer
216 R_adaptive_.segment(pos_num * TConfig::POS_MEASUREMENT_VECTOR_SIZE, TConfig::POS_MEASUREMENT_VECTOR_SIZE).setZero(); // NOLINT
217 } else {
218 // If no element is NaN, assign the covariance vector to the specified segment
219 R_adaptive_.segment(pos_num * TConfig::POS_MEASUREMENT_VECTOR_SIZE, TConfig::POS_MEASUREMENT_VECTOR_SIZE) = covariance; // NOLINT
220 }
221 // clang-format on
222 };

247 }

292 void declare_and_update_parameters(void)
293 {
294 // clang-format off
295 kalman_filter_params_.enable_mahalanobis_outlier_rejection = param_manager_->declare_and_get_value("enable.mahalanobis_outlier_rejection", false, tam::pmg::ParameterType::BOOL, "Use the Mahalanobis distance instead of fixed per-element bounds to reject measurement outliers").as_bool(); // NOLINT
296 kalman_filter_params_.enable_measurement_covariance_adaptation = param_manager_->declare_and_get_value("enable.measurement_covariance_adaptation", true, tam::pmg::ParameterType::BOOL, "Continuously adapt the measurement noise covariance R to the covariance reported by each measurement source").as_bool(); // NOLINT
297 kalman_filter_params_.covariance_adaption_limit = param_manager_->declare_and_get_value("kalman_filter.covariance_adaption.limit", 1.5, tam::pmg::ParameterType::DOUBLE, "Covariance value that R is raised to after a sudden jump, before it decays back toward the target covariance").as_double(); // NOLINT
298 kalman_filter_params_.covariance_adaption_decay = param_manager_->declare_and_get_value("kalman_filter.covariance_adaption.decay", 0.00165, tam::pmg::ParameterType::DOUBLE, "Amount by which R is reduced each update step while it decays from the limit back toward the target covariance").as_double(); // NOLINT
299 kalman_filter_params_.R_init = param_manager_->declare_and_get_value("kalman_filter.R_init", std::vector<double>(TConfig::MEASUREMENT_VECTOR_SIZE, 1.0), tam::pmg::ParameterType::DOUBLE_ARRAY, "Initial (and minimum) diagonal of the measurement noise covariance R of the EKF").as_double_array(); // NOLINT
300 kalman_filter_params_.Q = param_manager_->declare_and_get_value("kalman_filter.Q", std::vector<double>(TConfig::Q_VECTOR_SIZE, 1.0), tam::pmg::ParameterType::DOUBLE_ARRAY, "Process noise covariance Q diagonal of the EKF").as_double_array(); // NOLINT
301 kalman_filter_params_.outlier_bounds = param_manager_->declare_and_get_value("kalman_filter.outlier_bounds", std::vector<double>(TConfig::MEASUREMENT_VECTOR_SIZE, 1.0), tam::pmg::ParameterType::DOUBLE_ARRAY, "Per-element bounds used to reject measurement outliers when Mahalanobis rejection is disabled").as_double_array(); // NOLINT
302 previous_param_state_hash_ = param_manager_->get_state_hash();
303 // clang-format on
304 };
