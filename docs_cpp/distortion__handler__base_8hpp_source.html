43 void set_pose(const types::PoseStamped& pose)
44 {
45 // taken from
46 // https://github.com/autowarefoundation/autoware_universe/blob/main/sensing/autoware_cuda_pointcloud_preprocessor/src/cuda_pointcloud_preprocessor/cuda_pointcloud_preprocessor_node.cpp
47 // Set age of the history to 0.2s
48 double pose_history = 0.2; // seconds
49 while (!this->pose_history_.empty()) {
50 // rosbag replay
51 bool backwards_time_jump_detected = this->pose_history_.front().stamp > pose.stamp;
52 // Keep the last 0.2s of history
53 bool queue_exceeded = this->pose_history_.front().stamp < pose.stamp - pose_history * 1e9;
54
55 // Clean up pose history
56 if (backwards_time_jump_detected) {
57 this->pose_history_.clear();
58 } else if (queue_exceeded) {
59 this->pose_history_.pop_front();
60 } else {
61 break;
62 }
63 }
64
65 // Find sorted insert position
66 auto it = std::lower_bound(this->pose_history_.begin(), this->pose_history_.end(), pose,
67 [](const auto& history, const auto& pose) { return history.stamp < pose.stamp; });
68 // Don't add duplicate stamps
69 if (it != this->pose_history_.end() && it->stamp == pose.stamp) {
70 it->pose = pose.pose;
71 } else {
72 this->pose_history_.insert(it, pose);
73 }
74 }

144 float estimate_velocity() const
145 {
146 if (this->pose_history_.size() < 2) {
147 return 0.0f;
148 }
149 const auto& prev = this->pose_history_[this->pose_history_.size() - 2];
150 const auto& last = this->pose_history_.back();
151 const double dt = (static_cast<double>(last.stamp) - static_cast<double>(prev.stamp)) * 1.0e-9;
152 if (dt <= 0.0) {
153 return 0.0f;
154 }
155 return (last.pose.translation() - prev.pose.translation()).norm() / static_cast<float>(dt);
156 }

164 Sophus::SE3f interpolate_pose(const std::uint64_t stamp) const
165 {
166 const std::size_t n = this->pose_history_.size();
167 // Find the first pose with a stamp >= the requested stamp
168 auto it = std::lower_bound(this->pose_history_.begin(), this->pose_history_.end(), stamp,
169 [](const auto& history, const std::uint64_t s) { return history.stamp < s; });
170 // Select the bracketing pair, clamping to the ends for extrapolation
171 std::size_t i1 = static_cast<std::size_t>(std::distance(this->pose_history_.begin(), it));
172 if (i1 == 0) {
173 i1 = 1;
174 } else if (i1 >= n) {
175 i1 = n - 1;
176 }
177 const std::size_t i0 = i1 - 1;
178 const auto& p0 = this->pose_history_[i0];
179 const auto& p1 = this->pose_history_[i1];
180 // Interpolation factor (may lie outside [0, 1] when extrapolating)
181 const double denom = static_cast<double>(p1.stamp) - static_cast<double>(p0.stamp);
182 const float t =
183 denom > 0.0 ? static_cast<float>((static_cast<double>(stamp) - static_cast<double>(p0.stamp)) / denom) : 0.0f;
184 // Full SE(3) SLERP along the relative twist
185 return p0.pose * Sophus::SE3f::exp(t * (p0.pose.inverse() * p1.pose).log());
186 }

199 bool fit_polynom(const std::uint64_t frame_stamp, Eigen::Matrix<float, 6, types::POLYNOM_DEGREE + 1>& poly_coeff,
200 float& timestamp_offset)
201 {
202 // Check if we have enough poses in the range of the frame timestamp
203 bool valid_start = false;
204 bool valid_end = false;
205
206 // clang-format off
207 if (this->pose_history_.size() >= 2) {
208 for (size_t i = 0; i < this->pose_history_.size(); ++i) {
209 // Check start - assume that we need a pose at least 0.1s before the frame timestamp
210 if (!valid_start && std::abs(static_cast<double>(this->pose_history_[i].stamp) - static_cast<double>(frame_stamp - 0.1 * 1e9)) / 1.0e6 < this->config_.max_time_diff) { // NOLINT
211 valid_start = true;
212 }
213 // Check end
214 if (!valid_end && std::abs(static_cast<double>(this->pose_history_[i].stamp) - static_cast<double>(frame_stamp)) / 1.0e6 < this->config_.max_time_diff) { // NOLINT
215 valid_end = true;
216 }
217 }
218 }
219 // clang-format on
220 if (!valid_start || !valid_end) {
221 std::cout << "[DistortionHandler]: Could not interpolate!" << std::endl;
222 return false;
223 }
224 // (Inverse) reference pose at the frame timestamp
225 const Sophus::SE3f T_ref_inv = this->interpolate_pose(frame_stamp).inverse();
226 // Init variables
227 poly_coeff.setZero();
228 Eigen::MatrixXf knots(6, pose_history_.size());
229 Eigen::VectorXf times(pose_history_.size());
230
231 // Fill matrices
232 for (size_t i = 0; i < pose_history_.size(); ++i) {
233 // Compute time offset from frame timestamp to poses
234 // clang-format off
235 times(i) = static_cast<float>((static_cast<double>(pose_history_[i].stamp) - static_cast<double>(frame_stamp)) / 1.0e9); // NOLINT
236 // clang-format on
237 // Log-pose relative to the reference frame (se3 twist coordinates)
238 knots.col(i) = (T_ref_inv * pose_history_[i].pose).log().matrix();
239 }
240 // Normalize times
241 // Set timestamp offset to earliest pose used for interpolation
242 timestamp_offset = -times.minCoeff();
243 // Normalize times -> first pose at t=0
244 Eigen::VectorXf times_norm = times.array() + timestamp_offset;
245
246 // Fit polynomial (degree = config_.polynom_degree)
247 Eigen::MatrixXf A(times_norm.size(), types::POLYNOM_DEGREE + 1);
248 for (int i = 0; i < times_norm.size(); ++i) {
249 for (int j = 0; j <= types::POLYNOM_DEGREE; ++j) {
250 A(i, j) = std::pow(static_cast<float>(times_norm(i)), j);
251 }
252 }
253
254 // Solve least squares for each twist component and set coefficients
255 for (int k = 0; k < 6; ++k) {
256 Eigen::VectorXf y = knots.row(k).transpose();
257 poly_coeff.row(k) = A.householderQr().solve(y).transpose();
258 }
259 return true;
260 }
