32template <typename TConfig, typename T, typename FactorType, typename ReduceType>
33T reduce_correspondences(
34 std::vector<types::Correspondence<TConfig>>& correspondences, const FactorType factor, const ReduceType reduce)
35{
36 using correspondence_iterator = std::vector<types::Correspondence<TConfig>>::iterator;
37 const auto& reduced = tbb::parallel_reduce(
38 // Range
39 tbb::blocked_range<correspondence_iterator>{correspondences.begin(), correspondences.end()},
40 // Default initialize output
41 T{},
42 // 1st Lambda: Parallel computation
43 [&](const tbb::blocked_range<correspondence_iterator>& r, T element) -> T {
44 return std::transform_reduce(r.begin(), r.end(), element, reduce,
45 [&](types::Correspondence<TConfig>& correspondence) { return factor(correspondence); });
46 },
47 // 2nd Lambda: Parallel reduction of the private Jacboians
48 reduce);
49
50 return reduced;
51}

59template <typename TConfig, typename FactorType>
60types::LinearSystem build_linear_system(
61 std::vector<types::Correspondence<TConfig>>& correspondences, const FactorType& factor)
62{
63 auto sum_linear_systems = [](types::LinearSystem a, const types::LinearSystem& b) {
64 a.JTJ += b.JTJ;
65 a.JTr += b.JTr;
66 return a;
67 };
68 const types::LinearSystem& ls =
69 reduce_correspondences<TConfig, types::LinearSystem>(correspondences, factor, sum_linear_systems);
70 return ls;
71}

93template <typename TConfig>
94std::vector<types::Correspondence<TConfig>> get_correspondences(const std::vector<types::Point<TConfig>>& points,
95 const MapHandler<TConfig>* map, const float correspondence_threshold)
96{
97 using points_iterator = std::vector<types::Point<TConfig>>::const_iterator;
98 tbb::concurrent_vector<types::Correspondence<TConfig>> tbb_correspondences{};
99 tbb_correspondences.reserve(points.size());
100 float correspondence_threshold2 = correspondence_threshold * correspondence_threshold;
101
102 // calculate all closest neighbors in parallel
103 tbb::parallel_for(tbb::blocked_range<points_iterator>{points.cbegin(), points.cend()},
104 [&](const tbb::blocked_range<points_iterator>& r) {
105 std::for_each(r.begin(), r.end(), [&](const auto& point) {
106 const auto& correspondence = map->search_closest_neighbor(point, 1);
107 if (correspondence.distance < correspondence_threshold2) {
108 tbb_correspondences.emplace_back(correspondence);
109 }
110 });
111 });
112
113 // Combine thread-local results into a single std::vector
114 std::vector<types::Correspondence<TConfig>> correspondences(tbb_correspondences.begin(), tbb_correspondences.end());
115
116 return correspondences;
117}
