230 template <typename FactorType, typename ErrorType>
231#ifdef __CUDACC__
232 Sophus::SE3f solve(thrust::device_vector<types::Point<TConfig>>& frame, const MapHandler<TConfig>* map,
233 FactorType& factor, ErrorType& error, const float correspondence_threshold, const float kernel_scale,
234 ::cuda::stream_ref stream = {})
235#else
236 Sophus::SE3f solve(std::vector<types::Point<TConfig>>& frame, const MapHandler<TConfig>* map, FactorType& factor,
237 ErrorType& error, const float correspondence_threshold, const float kernel_scale)
238#endif
239 {
240 // Prepare factor and error
241 factor.set_kernel_scale(kernel_scale);
242 factor.set_correspondence_threshold(correspondence_threshold);
243 // Current transform is identity because we apply the initial guess to the source points
244 error.set_transform(Sophus::SE3f());
245 error.set_correspondence_threshold(correspondence_threshold);
246
247 // Call the solver
248 Sophus::SE3f T = Sophus::SE3f();
249 if (this->config_.solver_type == "GaussNewton") {
250 // Use Gauss-Newton method
251#ifdef __CUDACC__
252 T = this->solve_gaussnewton(frame, map, factor, error, stream);
253#else
254 T = this->solve_gaussnewton(frame, map, factor, error);
255#endif
256 } else if (this->config_.solver_type == "LevenbergMarquardt") {
257 // Use Levenberg-Marquardt method
258#ifdef __CUDACC__
259 T = this->solve_levenbergmarquardt(frame, map, factor, error, stream);
260#else
261 T = this->solve_levenbergmarquardt(frame, map, factor, error);
262#endif
263 } else {
264 throw std::runtime_error("Unknown solver type: " + this->config_.solver_type +
265 ". Supported types are GaussNewton and LevenbergMarquardt.");
266 }
267 return T;
268 }
277 template <typename FactorType, typename ErrorType>
278#ifdef __CUDACC__
279 Sophus::SE3f solve_gaussnewton(thrust::device_vector<types::Point<TConfig>>& frame, const MapHandler<TConfig>* map,
280 FactorType& factor, [[maybe_unused]] ErrorType& error, ::cuda::stream_ref stream = {})
281#else
282 Sophus::SE3f solve_gaussnewton(std::vector<types::Point<TConfig>>& frame, const MapHandler<TConfig>* map,
283 FactorType& factor, [[maybe_unused]] ErrorType& error)
284#endif
285 {
286 // Start timer
287 auto start = std::chrono::high_resolution_clock::now();
288
289 // Initialize variables
290 Sophus::SE3f T_icp = Sophus::SE3f();
291 types::SolverStatus solver_status = types::SolverStatus::NOT_CONVERGED;
292 std::int64_t iter = 0;
293 double time = 0.0;
294 for (std::int64_t j = 0; j < this->config_.max_iter; ++j) {
295 // Compute correspondences
296#ifdef __CUDACC__
297 map->search_closest_neighbor(frame, this->correspondences_device_, 1, stream);
298#else
299 std::vector<types::Correspondence<TConfig>> correspondences =
300 utils::get_correspondences(frame, map, factor.get_correspondence_threshold());
301#endif
302#ifdef __CUDACC__
303 if (this->correspondences_device_.empty()) break;
304#else
305 if (correspondences.empty()) break;
306#endif
307
308 // Build linear system
309#ifdef __CUDACC__
310 const types::LinearSystem& ls =
311 cuda::utils::build_linear_system(this->correspondences_device_, this->ls_final_d_, factor, stream);
312#else
313 const types::LinearSystem& ls = utils::build_linear_system(correspondences, factor);
314#endif
315 // Solve linear system
316 const Eigen::Matrix<float, 6, 1> dx =
317 (ls.JTJ + static_cast<float>(this->config_.damping_factor) * Eigen::Matrix<float, 6, 6>::Identity())
318 .ldlt()
319 .solve(-ls.JTr);
320 const Sophus::SE3f estimation = Sophus::SE3f::exp(dx);
321 // Transform points with new estimate
322#ifdef __CUDACC__
323 cuda::utils::transform_points(estimation, frame, stream);
324#else
325 utils::transform_points(estimation, frame);
326#endif
327 // Update transformation
328 T_icp = estimation * T_icp;
329 iter = j;
330 // Check solver status
331 solver_status = this->solver_state(dx);
332 // Compute elapsed time
333 // clang-format off
334 time = std::chrono::duration_cast<std::chrono::microseconds>(std::chrono::high_resolution_clock::now() - start).count() * 1.0e-3; // NOLINT
335 // clang-format on
336 // Check termination criteria
337 if (this->termination_criteria(solver_status, iter, time)) break;
338 }
339 // Update debug information
340 this->debug_.converged = solver_status == types::SolverStatus::CONVERGED;
341 this->debug_.registration_time = time;
342 this->debug_.damping_factor = static_cast<float>(this->config_.damping_factor);
343 this->debug_.num_iter = iter;
344 this->debug_.num_points_frame = frame.size();
345 return T_icp;
346 }
355 template <typename FactorType, typename ErrorType>
356#ifdef __CUDACC__
357 Sophus::SE3f solve_levenbergmarquardt(thrust::device_vector<types::Point<TConfig>>& frame,
358 const MapHandler<TConfig>* map, FactorType& factor, ErrorType& error, ::cuda::stream_ref stream = {})
359#else
360 Sophus::SE3f solve_levenbergmarquardt(
361 std::vector<types::Point<TConfig>>& frame, const MapHandler<TConfig>* map, FactorType& factor, ErrorType& error)
362#endif
363 {
364 // Start timer
365 auto start = std::chrono::high_resolution_clock::now();
366
367 // Initialize variables
368 Sophus::SE3f T_icp = Sophus::SE3f();
369 types::SolverStatus solver_status = types::SolverStatus::NOT_CONVERGED;
370 std::int64_t iter = 0;
371 std::int64_t inner_iter = 0;
372 float current_error = 0.0f;
373 // Initial lambda for Levenberg-Marquardt -> overwrite before first use
374 float current_lambda = 0.0f;
375 double time = 0.0;
376 // Outer loop
377 for (std::int64_t j = 0; j < this->config_.max_iter; ++j) {
378 // Compute correspondences
379#ifdef __CUDACC__
380 map->search_closest_neighbor(frame, this->correspondences_device_, 1, stream);
381#else
382 std::vector<types::Correspondence<TConfig>> correspondences =
383 utils::get_correspondences(frame, map, factor.get_correspondence_threshold());
384#endif
385#ifdef __CUDACC__
386 if (this->correspondences_device_.empty()) break;
387#else
388 if (correspondences.empty()) break;
389#endif
390 // Reset error transform to identity for the current iteration
391 error.set_transform(Sophus::SE3f());
392 // Build linear system and compute current error
393#ifdef __CUDACC__
394 const types::LinearSystem& ls =
395 cuda::utils::build_linear_system(this->correspondences_device_, this->ls_final_d_, factor, stream);
396 current_error = cuda::utils::compute_error(this->correspondences_device_, this->error_d_, error, stream);
397#else
398 const types::LinearSystem& ls = utils::build_linear_system(correspondences, factor);
399 current_error = utils::compute_error(correspondences, error);
400#endif
401
402 // Init Levenberg-Marquardt parameters
403 Sophus::SE3f delta = Sophus::SE3f();
404 Sophus::SE3f delta_lm = Sophus::SE3f();
405 // Set initial lambda for first iteration
406 if (j == 0) {
407 current_lambda = std::max(
408 static_cast<float>(this->config_.damping_factor) * ls.JTJ.diagonal().array().abs().maxCoeff(), 1.0e-12f);
409 }
410 // Start Levenberg-Marquardt inner loop
411 bool success_lm = false;
412 float nu = static_cast<float>(this->config_.damping_scale);
413 for (std::int64_t lm_iter = 0; lm_iter < this->config_.max_inner_iter; ++lm_iter) {
414 // Solve linear system
415 const Eigen::Matrix<float, 6, 1> candidate_dx =
416 (ls.JTJ + current_lambda * Eigen::Matrix<float, 6, 6>::Identity()).ldlt().solve(-ls.JTr); // NOLINT
417 delta_lm = Sophus::SE3f::exp(candidate_dx);
418 // Set the transformation to the factor
419 error.set_transform(delta_lm);
420 // Compute the error for the new transform
421#ifdef __CUDACC__
422 float candidate_error =
423 cuda::utils::compute_error(this->correspondences_device_, this->error_d_, error, stream);
424#else
425 float candidate_error = utils::compute_error(correspondences, error);
426#endif
427 float rho = (current_error - candidate_error) / (candidate_dx.dot(current_lambda * candidate_dx - ls.JTr));
428
429 // Update debug information
430 inner_iter = lm_iter;
431 types::SolverStatus solver_status_candidate = this->solver_state(candidate_dx);
432 // Termination criteria inner loop
433 if (rho < 0.0f) {
434 if (solver_status_candidate == types::SolverStatus::CONVERGED) {
435 solver_status = solver_status_candidate;
436 success_lm = true;
437 break;
438 }
439 // Increase the damping factor otherwise
440 // (always ensure that lambda can never become nummerically unstable)
441 current_lambda = std::max(nu * current_lambda, 1.0e-12f);
442 nu *= 2.0f;
443 // Jump to the next iteration of the inner loop
444 continue;
445 }
446 // Update outer loop variables if rho > 0
447 T_icp = delta_lm * T_icp; // Update complete transformation
448 delta = delta_lm; // Update delta to transform points
449 solver_status = solver_status_candidate; // Update solver status
450 current_lambda =
451 std::max(current_lambda * std::max(1.0f / 3.0f, 1.0f - std::pow(2.0f * rho - 1.0f, 3.0f)), 1.0e-12f);
452 success_lm = true;
453 break; // Break the inner loop if we found a valid solution
454 }
455 // Break outer loop if inner loop did not succeed
456 if (!success_lm) break;
457 // Transform points with new estimate
458#ifdef __CUDACC__
459 cuda::utils::transform_points(delta, frame, stream);
460#else
461 utils::transform_points(delta, frame);
462#endif
463 // Termination criteria
464 iter = j;
465 // Compute elapsed time
466 // clang-format off
467 time = std::chrono::duration_cast<std::chrono::microseconds>(std::chrono::high_resolution_clock::now() - start).count() * 1.0e-3; // NOLINT
468 // clang-format on
469 if (this->termination_criteria(solver_status, iter, time)) break;
470 }
471 // Update debug information
472 this->debug_.converged = solver_status == types::SolverStatus::CONVERGED;
473 this->debug_.registration_time = time;
474 this->debug_.damping_factor = current_lambda;
475 this->debug_.conditional["error"] = static_cast<double>(current_error);
476 this->debug_.num_iter = iter;
477 this->debug_.conditional["num_inner_iter"] = static_cast<std::int64_t>(inner_iter);
478 this->debug_.num_points_frame = frame.size();
479 return T_icp;
480 }
481
487 types::SolverStatus solver_state(const Eigen::Matrix<float, 6, 1>& dx) const
488 {
489 const double dx_norm = dx.norm();
490 // Check if an actual solution was computed
491 // As the check against the correspondence threshold is done in the linear system build,
492 // it could happen that there is no valid correspondence found, thus the linear system
493 // consists only of 0 and dx_norm = 0
494 if (dx_norm <= 1.0e-12) return types::SolverStatus::INVALID;
495 // Check convergence
496 if (dx_norm < this->config_.convergence_criterion) {
497 return types::SolverStatus::CONVERGED;
498 } else {
499 return types::SolverStatus::NOT_CONVERGED;
500 }
501 }
502
510 bool termination_criteria(const types::SolverStatus& status, const std::int64_t iter, const double time) const
511 {
512 // clang-format off
513 if (status == types::SolverStatus::CONVERGED || status == types::SolverStatus::INVALID) return true; // NOLINT
514 // Check if the maximum number of iterations is reached
515 if (iter >= this->config_.max_iter) return true;
516 // Check if the maximum time is exceeded
517 if (time >= this->config_.max_time) return true;
518 // If none of the criteria are met, continue optimization
519 return false;
520 // clang-format on
521 }
