33template <typename TConfig>
34std::unique_ptr<tam::core::state::MapHandler<TConfig>> init_map(
35 const size_t num_points = 10, const std::string& cov_reg = "SVD")
36{
37 // Construct from config and debug objects
38 tam::core::state::types::MapConfig config;
39 tam::core::state::types::MapDebug debug;
40 config.voxel_size = 1.0;
41 config.max_distance = 50.0;
42 config.cov_regularization = cov_reg; // Options: "SVD", "FROBENIUS", "MIN_EIGENVALUE"
43#ifdef __CUDACC__
44 auto map_ = tam::core::state::cuda::VoxelHashMap<TConfig>::from_config(config, debug);
45#else
46 auto map_ = tam::core::state::VoxelHashMap<TConfig>::from_config(config, debug); // NOLINT
47#endif
48 map_->init();
49
50 std::vector<tam::core::state::types::Point<tam::core::state::types::Point_XYZ>> frame{};
51 if (num_points <= 10) {
52 // Define a frame of points manually (for deterministic testing of neighbor search)
53 tam::core::state::types::Point<tam::core::state::types::Point_XYZ> point_1, point_2, point_3, point_4, point_5,
54 point_6, point_7, point_8, point_9, point_10;
55 point_1.pos = Eigen::Vector3f(5.1, 2.1, 3.1);
56 point_2.pos = Eigen::Vector3f(5.8, 2.5, 3.9);
57 point_3.pos = Eigen::Vector3f(5.9, 2.9, 3.9);
58 point_4.pos = Eigen::Vector3f(4.1, 1.0, 4.0);
59 point_5.pos = Eigen::Vector3f(4.2, 1.1, 3.0);
60 point_6.pos = Eigen::Vector3f(6.3, 3.2, 3.1);
61 point_7.pos = Eigen::Vector3f(5.4, 2.3, 4.2);
62 point_8.pos = Eigen::Vector3f(5.5, 2.4, 2.3);
63 point_9.pos = Eigen::Vector3f(5.6, 2.5, 4.4);
64 point_10.pos = Eigen::Vector3f(5.7, 2.6, 2.5);
65 std::vector<tam::core::state::types::Point<tam::core::state::types::Point_XYZ>> frame_man{
66 point_1, point_2, point_3, point_4, point_5, point_6, point_7, point_8, point_9, point_10};
67 frame = frame_man;
68 } else {
69 // Generate random points within one voxel
70 frame.reserve(num_points);
71 std::random_device rd;
72 std::mt19937 gen(rd());
73 std::uniform_real_distribution<> dis(-config.voxel_size / 2, config.voxel_size / 2);
74
75 for (size_t i = 0; i < num_points; ++i) {
76 tam::core::state::types::Point<tam::core::state::types::Point_XYZ> point;
77 point.pos = Eigen::Vector3f(0.5f + dis(gen), 0.5f + dis(gen), 0.5f + dis(gen));
78 frame.push_back(point);
79 }
80 }
81 // Add points to the map
82 map_->add_points(frame);
83 return map_;
84}
