36 explicit OdometryNode(const rclcpp::NodeOptions& options)
37 : NodeBase<TConfig, types::OdometryNodeConfig, types::OdometryNodeDebug>(options)
38 {
39 // Call common initialization here to be able to call set_config/set_logging,
40 // which are pure virtual in the base class
41 this->init_node_common();
42 // Initialize node and pipeline if not using external guess
43 if constexpr (TConfig::MODEL != types::ModelType::EXTERNALGUESS) {
44 // Set member variable for initialization to true as waiting for odometry is not necessary
45 this->init_status_ = types::InitStatus::READY;
46 // Initialize model
47 this->pipeline_->init_model_config();
48 // Initialize dynamic transform broadcaster
49 this->tf_broadcaster_ = std::make_unique<tf2_ros::TransformBroadcaster>(*this);
50 }
51
52 // Finish
53 this->monitor_->initialization_finished();
54 RCLCPP_INFO(this->get_logger(), "[%s]: Node initialized!", this->get_name()); // NOLINT
55 }

59private:
60 void timer_callback() override
61 {
62 // Check for timeouts
63 this->topic_watchdog_->check_timeouts();
64 // Check initial guess status
65 if (this->init_status_ == types::InitStatus::WAITING_FOR_EKF && this->config_.wait_tf) {
66 std::string tf_error{};
67 if (this->tf_buffer_->canTransform(this->config_.odom_frame, this->config_.child_frame, tf2::TimePointZero,
68 tf2::durationFromSec(0.0), &tf_error)) {
69 const types::PoseStamped pose = utils::transform2pose(this->tf_buffer_->lookupTransform(
70 this->config_.odom_frame, this->config_.child_frame, tf2::TimePointZero)); // NOLINT
71 this->pipeline_->set_pose_model(pose, true);
72 // Finish initialization
73 RCLCPP_INFO(this->get_logger(), "[%s]: Handshake successful!", this->get_name());
74 this->monitor_->set_message("Initialized");
75 this->monitor_->set_error_lvl("handshake", tam::types::ErrorLvl::OK);
76 this->init_status_ = types::InitStatus::READY;
77 } else {
78 this->monitor_->set_message("Waiting for handshake with EKF");
79 this->monitor_->set_error_lvl("handshake", tam::types::ErrorLvl::WARN);
80 RCLCPP_WARN(this->get_logger(), "[%s]: Waiting for handshake with EKF", this->get_name());
81 }
82 // Send a status to the dashboard to indicate the initialization status
83 this->monitor_->update();
84 }

93 void cloud_callback(const sensor_msgs::msg::PointCloud2::ConstSharedPtr msg_ptr) override
94 {
95 // Return if not initialized
96 if (this->init_status_ != types::InitStatus::READY && this->config_.wait_tf) {
97 return;
98 }
99
100 // Start timer
101 auto start = std::chrono::high_resolution_clock::now();
102
103 // clang-format off
104 // Convert message to points
105 if (msg_ptr->header.frame_id != this->config_.cloud_frame) {
106 RCLCPP_ERROR(this->get_logger(), "[%s]: Frame id of pointcloud differs from specified one!", this->get_name()); // NOLINT
107 }
108 std::vector<types::Point<TConfig>> frame = utils::cloud2eigen<TConfig>(*msg_ptr, this->pc_transform_); // NOLINT
109 // Fetch initial guess and set it to pipeline if configured
110 if constexpr (TConfig::MODEL == types::ModelType::EXTERNALGUESS) {
111 types::PoseStamped init_guess{};
112 bool valid_guess{true};
113 std::string tf_error;
114 // Try to fetch initial guess from tf
115 if (this->tf_buffer_->canTransform(this->config_.odom_frame, this->config_.child_frame, msg_ptr->header.stamp, tf2::durationFromSec(0.0), &tf_error)) { // NOLINT
116 init_guess = utils::transform2pose(this->tf_buffer_->lookupTransform(this->config_.odom_frame, this->config_.child_frame, msg_ptr->header.stamp)); // NOLINT
117 } else if (this->tf_buffer_->canTransform(this->config_.odom_frame, this->config_.child_frame, tf2::TimePointZero, tf2::durationFromSec(0.0), &tf_error)) { // NOLINT
118 init_guess = utils::transform2pose(this->tf_buffer_->lookupTransform(this->config_.odom_frame, this->config_.child_frame, tf2::TimePointZero)); // NOLINT
119 } else {
120 RCLCPP_WARN(this->get_logger(), "[%s]: Could not lookup initial guess available from tf!", this->get_name()); // NOLINT
121 const auto odom = this->pipeline_->get_odometry();
122 init_guess.pose = odom.pose.pose;
123 init_guess.stamp = utils::stamp2ns(msg_ptr->header.stamp);
124 valid_guess = false;
125 }
126 this->pipeline_->set_pose_model(init_guess, valid_guess);
127 }
128 // clang-format on
129 // Set deskewing poses to the pipeline by sampling tf around the frame stamp
130 this->set_undistortion_poses(utils::stamp2ns(msg_ptr->header.stamp));
131 // Register frame
132 types::Odometry odom = this->pipeline_->register_frame(frame, utils::stamp2ns(msg_ptr->header.stamp));
133
134 // Compute elapsed time
135 // clang-format off
136 double time = std::chrono::duration_cast<std::chrono::microseconds>(std::chrono::high_resolution_clock::now() - start).count() * 1.0e-3; // NOLINT
137 // clang-format on
138 this->debug_.callback_time = time;
139 // Initialize if the pose is valid
140 if (odom.status.level == types::DiagnosticLevel::OK) {
141 this->monitor_->set_error_lvl("handshake", tam::types::ErrorLvl::OK);
142 this->init_status_ = types::InitStatus::READY;
143 }
144 // Publish diagnostics
145 utils::set_monitor(odom, this->monitor_.get());
146 const nav_msgs::msg::Odometry odom_msg = utils::odom2msg(odom, this->config_.odom_frame, this->config_.child_frame);
147 // Publish status and odometry with same stamp
148 this->monitor_->update(odom_msg.header);
149 this->odom_publisher_->publish(odom_msg);
150 // Publish tf if no external guess
151 if constexpr (TConfig::MODEL != types::ModelType::EXTERNALGUESS) {
152 geometry_msgs::msg::TransformStamped transform_msg;
153 transform_msg.header.stamp = msg_ptr->header.stamp;
154 transform_msg.header.frame_id = this->config_.odom_frame;
155 transform_msg.child_frame_id = this->config_.child_frame;
156 transform_msg.transform = utils::sophus2transform(odom.pose.pose);
157 tf_broadcaster_->sendTransform(transform_msg);
158 }
159 // Publish debug signals
160 this->tsl_publisher_->trigger();
161 // Publish frame if debug mode is enabled
162 if (this->pipeline_->get_config().debug_mode) {
163 // Publish frame
164 std_msgs::msg::Header cloud_header = msg_ptr->header;
165 cloud_header.frame_id = this->config_.child_frame;
166 this->frame_publisher_->publish(utils::eigen2cloud<TConfig>(this->pipeline_->get_frame(), cloud_header));
167 // Publish map
168 if (this->pipeline_->get_config().update_map) {
169 cloud_header.frame_id = this->config_.odom_frame;
170 this->map_publisher_->publish(utils::eigen2cloud<TConfig>(this->pipeline_->get_map(), cloud_header));
171 }
172 }
173 }
