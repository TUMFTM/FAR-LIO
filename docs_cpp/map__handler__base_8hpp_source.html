335 __host__ void set_normal_covariance(thrust::device_vector<types::Neighbors<TConfig>>& neighbors,
336 const int16_t num_neighbors, const size_t num_multiprocessors, ::cuda::stream_ref stream = {})
337 requires types::HASNORMALCOV<TConfig>
338 {
339 nvtxRangePush("set_normal_covariance");
340 auto grid_size = std::min(4 * num_multiprocessors, (neighbors.size() + cuda::BLOCK_SIZE - 1) / cuda::BLOCK_SIZE);
341
342 types::Neighbors<TConfig>* raw_neighbors = thrust::raw_pointer_cast(neighbors.data());

348 nvtxRangePop();

350#else
351 void set_normal_covariance(types::Neighbors<TConfig>& neighbor, const int16_t num_neighbors)
352 requires types::HASNORMALCOV<TConfig>
353 {
354 // Check if enough neighbors were found and set the normal to zero if not
355 if (neighbor.num_neighbors < num_neighbors) {
356 neighbor.point->normal = Eigen::Vector3f::Zero();
357 neighbor.point->cov = Eigen::Matrix3f::Zero();
358 return;
359 }
360 Eigen::Vector3f sum_pts = Eigen::Vector3f::Zero();
361 Eigen::Matrix3f sum_pts_cov = Eigen::Matrix3f::Zero();
362
363 // clang-format off
364 for (size_t i = 0; i < neighbor.num_neighbors; ++i) sum_pts += neighbor.neighbor[i]->pos;
365 const Eigen::Vector3f mean = sum_pts / static_cast<float>(neighbor.num_neighbors);
366 for (size_t i = 0; i < neighbor.num_neighbors; ++i)
367 sum_pts_cov += (neighbor.neighbor[i]->pos - mean) * (neighbor.neighbor[i]->pos - mean).transpose(); // NOLINT
368 const Eigen::Matrix3f cov = sum_pts_cov / static_cast<float>(neighbor.num_neighbors);
369 // clang-format on
370 // Compute the normal
371 Eigen::SelfAdjointEigenSolver<Eigen::Matrix3f> eigensolver;
372 eigensolver.computeDirect(cov);
373
374 // Set normal and covariance
375 const Eigen::Vector3f normal = eigensolver.eigenvectors().col(0).normalized();
376 if (neighbor.point->pos.dot(normal) > 0) {
377 neighbor.point->normal = -normal;
378 } else {
379 neighbor.point->normal = normal;
380 }
386 switch (this->cov_regularization_type_) {
387 case types::CovRegularizationType::SVD: {
388 // SVD regularization
389 const Eigen::Vector3f values(1.0e-3, 1.0, 1.0);
390 neighbor.point->cov = eigensolver.eigenvectors() * values.asDiagonal() * eigensolver.eigenvectors().transpose();
391 break;
392 }
393 case types::CovRegularizationType::FROBENIUS: {
394 // Frobenius regularization
395 float lambda = 1e-3;
396 Eigen::Matrix3f C = cov + lambda * Eigen::Matrix3f::Identity();
397 Eigen::Matrix3f C_inv = C.inverse();
398 Eigen::Matrix3f C_norm = (C_inv / C_inv.norm()).inverse();
399 neighbor.point->cov = C_norm;
400 break;
401 }
402 case types::CovRegularizationType::MIN_EIGENVALUE: {
403 // Min eigenvalue regularization
404 const Eigen::Vector3f eigen_values = eigensolver.eigenvalues();
405 const Eigen::Matrix3f eigen_vectors = eigensolver.eigenvectors();
406
407 // Create regularized eigenvalues
408 Eigen::Vector3f regularized_values;
409 regularized_values[0] = fmaxf(1.0e-3, eigen_values[0]);
410 regularized_values[1] = fmaxf(1.0e-3, eigen_values[1]);
411 regularized_values[2] = fmaxf(1.0e-3, eigen_values[2]);
412
413 // Reconstruct regularized covariance directly
414 neighbor.point->cov = eigen_vectors * regularized_values.asDiagonal() * eigen_vectors.transpose();
415 break;
416 }
417 default: {
418 // Default case: use the identity matrix
419 neighbor.point->cov = Eigen::Matrix3f::Identity();
420 }
421 }
422 return;
423 }
