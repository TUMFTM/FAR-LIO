77 const Eigen::Vector<double, TConfig::INPUT_VECTOR_SIZE> update_input_vector(
78 const Eigen::Ref<const Eigen::Vector<double, TConfig::INPUT_VECTOR_SIZE*(TConfig::NUM_IMU_MEASUREMENT + TConfig::NUM_BACKUP_IMU_MEASUREMENT)>>& u_raw, // NOLINT
79 const Eigen::Ref<const Eigen::Vector<double, TConfig::INPUT_VECTOR_SIZE*(TConfig::NUM_IMU_MEASUREMENT + TConfig::NUM_BACKUP_IMU_MEASUREMENT)>>& u_fusion_vec, // NOLINT
80 std::optional<Eigen::Vector<double, TConfig::INPUT_VECTOR_SIZE*(TConfig::NUM_IMU_MEASUREMENT + TConfig::NUM_BACKUP_IMU_MEASUREMENT)>> u_cov = std::nullopt) // NOLINT
81 // clang-format on
82 {
83 if (param_manager_->get_state_hash() != previous_param_state_hash_) {
84 declare_and_update_parameters();
85 }
86 if (!u_cov.has_value()) {
87 u_cov.emplace();
88 u_cov->setZero();
89 }
90
91 // average the imu data
92 Eigen::Vector<double, TConfig::INPUT_VECTOR_SIZE> u = Eigen::Vector<double, TConfig::INPUT_VECTOR_SIZE>::Zero();
93 const Eigen::Vector<double, TConfig::INPUT_VECTOR_SIZE> u_average = average_input_vector(u_raw, u_fusion_vec, *u_cov);
94
95 // update the imu bias FiFo buffer
96 if (update_bias_) {
97 Eigen::Vector<double, TConfig::INPUT_VECTOR_SIZE> u_outlier =
98 Eigen::Vector<double, TConfig::INPUT_VECTOR_SIZE>::Zero();
99 if (abs(u_average[TConfig::INPUT_DPSI_RADPS]) < 0.05)
100 u_outlier[TConfig::INPUT_DPSI_RADPS] = u_average[TConfig::INPUT_DPSI_RADPS];
101 if (abs(u_average[TConfig::INPUT_AX_MPS2]) < 1.0)
102 u_outlier[TConfig::INPUT_AX_MPS2] = u_average[TConfig::INPUT_AX_MPS2];
103 if (abs(u_average[TConfig::INPUT_AY_MPS2]) < 1.0)
104 u_outlier[TConfig::INPUT_AY_MPS2] = u_average[TConfig::INPUT_AY_MPS2];
105
106 if constexpr (TConfig::INPUT_VECTOR_SIZE == 6) {
107 if (abs(u_average[TConfig::INPUT_DPHI_RADPS]) < 0.05)
108 u_outlier[TConfig::INPUT_DPHI_RADPS] = u_average[TConfig::INPUT_DPHI_RADPS];
109 if (abs(u_average[TConfig::INPUT_DTHETA_RADPS]) < 0.05)
110 u_outlier[TConfig::INPUT_DTHETA_RADPS] = u_average[TConfig::INPUT_DTHETA_RADPS];
111 if (abs(u_average[TConfig::INPUT_AZ_MPS2]) < 1.0)
112 u_outlier[TConfig::INPUT_AZ_MPS2] = u_average[TConfig::INPUT_AZ_MPS2];
113 }
114
115 sum_bias_ += u_outlier;
116 if (u_bias_fifo_.size() > 10000) {
117 sum_bias_ -= u_bias_fifo_.back();
118 u_bias_fifo_.pop_back();
119 }
120 u_bias_fifo_.push_front(u_outlier);
121 imu_bias_ = sum_bias_ / static_cast<double>(u_bias_fifo_.size());
122
123 imu_bias_[TConfig::INPUT_AX_MPS2] = initial_imu_bias_[TConfig::INPUT_AX_MPS2];
124 imu_bias_[TConfig::INPUT_AY_MPS2] = initial_imu_bias_[TConfig::INPUT_AY_MPS2];
125 if constexpr (TConfig::INPUT_VECTOR_SIZE == 6) {
126 imu_bias_[TConfig::INPUT_AZ_MPS2] = initial_imu_bias_[TConfig::INPUT_AZ_MPS2];
127 }
128 }
129
130 if constexpr (TConfig::INPUT_VECTOR_SIZE == 6) {
131 // subtract estimated sensor bias (for the esekf (SOLA. equation 259))
132 u = u_average - imu_bias_;
133 } else if constexpr (TConfig::INPUT_VECTOR_SIZE == 3) {
134 u = u_average;
135 // clamp the banking angle to a physically valid range so 1 / cos(banking) and tan(banking)
136 // cannot diverge (a real track never banks anywhere near +-90 deg)
137 const double banking_rad = std::clamp(-external_orientation_.x, -1.5, 1.5);
138 // compensate the banking angle for dpsi_rad
139 u[TConfig::INPUT_DPSI_RADPS] /= std::cos(banking_rad);
140 // add ay_calibration_offset to all ay_mps2 measurements
141 u[TConfig::INPUT_AY_MPS2] -= imu_bias_[TConfig::INPUT_AY_MPS2];
142 // compensate banking for ay_mps2
143 u[TConfig::INPUT_AY_MPS2] *= std::sin(banking_rad) * std::tan(banking_rad) + std::cos(banking_rad);
144 // compensate gravity for ay_mps2
145 u[TConfig::INPUT_AY_MPS2] += std::tan(banking_rad) * tam::constants::g_earth;
146 } else {
147 static_assert(
148 TConfig::INPUT_VECTOR_SIZE == 3 || TConfig::INPUT_VECTOR_SIZE == 6, "[IMUHandler]: invalid INPUT_VECTOR_SIZE");
149 }
150 return u;
151 }

161 const Eigen::Vector<double, TConfig::INPUT_VECTOR_SIZE> filter_imu_measurements(
162 const Eigen::Ref<const Eigen::Vector<double, TConfig::INPUT_VECTOR_SIZE>>& u_raw, const double imu_num)
163 {
164 if (initialize_filter_) {
165 for (int i = 0; i < TConfig::NUM_IMU_MEASUREMENT + TConfig::NUM_BACKUP_IMU_MEASUREMENT; ++i) {
166 // set the Filter Coefficients
167 const Eigen::VectorXd imu_filter_coefficients_imu = Eigen::Map<Eigen::VectorXd>(
168 imu_handler_params_.filter_coefficients[i].data(), imu_handler_params_.filter_coefficients[i].size());
169 for (int vec_pos = 0; vec_pos < TConfig::INPUT_VECTOR_SIZE; ++vec_pos) {
170 filter_imu_[i * TConfig::INPUT_VECTOR_SIZE + vec_pos] =
171 std::make_unique<tam::core::state::FIR>(imu_filter_coefficients_imu);
172 }
173 }
174 initialize_filter_ = false;
175 }
176
177 Eigen::Vector<double, TConfig::INPUT_VECTOR_SIZE> u_filt;
178 u_filt.setZero();
179 // filter all values
180 for (int i = 0; i < TConfig::INPUT_VECTOR_SIZE; ++i) {
181 u_filt[i] = filter_imu_[imu_num * TConfig::INPUT_VECTOR_SIZE + i]->step(u_raw[i]);
182 }
183 return u_filt;
184 }

249 // clang-format off
250 const Eigen::Vector<double, TConfig::INPUT_VECTOR_SIZE> average_input_vector(
251 const Eigen::Ref<const Eigen::Vector<double, TConfig::INPUT_VECTOR_SIZE*(TConfig::NUM_IMU_MEASUREMENT + TConfig::NUM_BACKUP_IMU_MEASUREMENT)>>& u_raw, // NOLINT
252 const Eigen::Ref<const Eigen::Vector<double, TConfig::INPUT_VECTOR_SIZE*(TConfig::NUM_IMU_MEASUREMENT + TConfig::NUM_BACKUP_IMU_MEASUREMENT)>>& u_fusion_vec, // NOLINT
253 const Eigen::Ref<const Eigen::Vector<double, TConfig::INPUT_VECTOR_SIZE*(TConfig::NUM_IMU_MEASUREMENT + TConfig::NUM_BACKUP_IMU_MEASUREMENT)>>& u_cov) // NOLINT
254 // clang-format on
255 {
256 // Configurable constants
257 constexpr double EPSILON = 1e-6; // Minimum allowed covariance
258 constexpr double MAX_COV = 10.0; // Maximum allowed covariance
259
260 // clang-format off
261 Eigen::Vector<double, TConfig::INPUT_VECTOR_SIZE> u_sum = Eigen::Vector<double, TConfig::INPUT_VECTOR_SIZE>::Zero(); // NOLINT
262 Eigen::Vector<double, TConfig::INPUT_VECTOR_SIZE> weight_sum = Eigen::Vector<double, TConfig::INPUT_VECTOR_SIZE>::Zero(); // NOLINT
263 Eigen::Vector<double, TConfig::INPUT_VECTOR_SIZE> u = Eigen::Vector<double, TConfig::INPUT_VECTOR_SIZE>::Zero(); // NOLINT
264 // clang-format on
265
266 for (int i = 0;
267 i < TConfig::INPUT_VECTOR_SIZE * (TConfig::NUM_IMU_MEASUREMENT + TConfig::NUM_BACKUP_IMU_MEASUREMENT); ++i) {
268 if (u_fusion_vec[i] > 0.0) { // Valid measurement
269 // Clamp covariance between EPSILON and MAX_COV
270 const double clamped_cov = std::min(std::max(u_cov[i], EPSILON), MAX_COV);
271 const double weight = 1.0 / clamped_cov;
272 u_sum[i % TConfig::INPUT_VECTOR_SIZE] += u_raw[i] * weight;
273 weight_sum[i % TConfig::INPUT_VECTOR_SIZE] += weight;
274 }
275 }
276
277 // take the average of the IMU measurements if a valid measurement was received
278 for (int i = 0; i < TConfig::INPUT_VECTOR_SIZE; ++i) {
279 if (weight_sum[i] > 0.0) u[i] = u_sum[i] / weight_sum[i];
280 }
281
282 return u;
283 }
