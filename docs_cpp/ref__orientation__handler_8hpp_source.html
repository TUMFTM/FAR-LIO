71 const tam::types::control::Odometry& update(
72 const Eigen::Ref<const Eigen::Vector<double, TConfig::STATE_VECTOR_SIZE>>& x,
73 const Eigen::Ref<const Eigen::Vector<double, TConfig::INPUT_VECTOR_SIZE>>& u)
74 requires(!std::is_same_v<TConfig, tam::core::state::EKF_2D>)
75 {
76 if (param_manager_->get_state_hash() != previous_param_state_hash_) {
77 this->declare_and_update_parameters();
78 }
79 // initialize the numerical differentiation
80 if (first_iteration_) {
81 v_t_minus_two_ = x.segment(TConfig::STATE_VX_MPS, TConfig::VEL_MEASUREMENT_VECTOR_SIZE);
82
83 // set the Filter Coefficients
84 const Eigen::VectorXd imu_filter_coefficients =
85 Eigen::Map<Eigen::VectorXd>(ref_orientation_handler_params_.v_dot_filter_coefficients.data(),
86 ref_orientation_handler_params_.v_dot_filter_coefficients.size());
87
88 for (std::size_t i = 0; i < TConfig::VEL_MEASUREMENT_VECTOR_SIZE; ++i) {
89 filter_[i] = std::make_unique<tam::core::state::FIR>(imu_filter_coefficients);
90 }
91
92 first_iteration_ = false;
93 return reference_orientation_;
94 }
95 if (second_iteration_) {
96 v_t_minus_one_ = x.segment(TConfig::STATE_VX_MPS, TConfig::VEL_MEASUREMENT_VECTOR_SIZE);
97 second_iteration_ = false;
98 return reference_orientation_;
99 }
100
101 // compute the central derivative of the vehicle velocity
102 Eigen::Vector<double, TConfig::VEL_MEASUREMENT_VECTOR_SIZE> v_dot{};
103 v_dot =
104 (x.segment(TConfig::STATE_VX_MPS, TConfig::VEL_MEASUREMENT_VECTOR_SIZE) - v_t_minus_two_) / (2 * TConfig::TS);
105
106 // buffer the vehicle velocities for the next cycle
107 v_t_minus_two_ = v_t_minus_one_;
108 v_t_minus_one_ = x.segment(TConfig::STATE_VX_MPS, TConfig::VEL_MEASUREMENT_VECTOR_SIZE);
109
110 // FIR Filter the velocity derivative
111 for (int i = 0; i < TConfig::VEL_MEASUREMENT_VECTOR_SIZE; ++i) {
112 v_dot[i] = filter_[i]->step(v_dot[i]);
113 }
114
115 // calculate the reference angles base on:
116 // https://acl.kaist.ac.kr/wp-content/uploads/2021/10/2013partd_OJW.pdf
117 // to ensure that no inf is returned by the asin the values have to be clamped
118 const double phi_value = (-v_dot[TConfig::MEASUREMENT_VY_MPS] + u[TConfig::INPUT_AY_MPS2] -
119 u[TConfig::INPUT_DPSI_RADPS] * x[TConfig::STATE_VX_MPS] +
120 u[TConfig::INPUT_DPHI_RADPS] * x[TConfig::STATE_VZ_MPS]) /
121 (tam::constants::g_earth * std::cos(x[TConfig::STATE_THETA_RAD]));
122
123 // clamp bounds the +-inf case; guard the 0/0 = NaN case so asin never propagates a NaN
124 reference_orientation_.orientation_rad.x =
125 std::asin(std::clamp(std::isnan(phi_value) ? 0.0 : phi_value, -0.5, 0.5));
126
127 const double theta_value = (v_dot[TConfig::MEASUREMENT_VX_MPS] - u[TConfig::INPUT_AX_MPS2] -
128 u[TConfig::INPUT_DPSI_RADPS] * x[TConfig::STATE_VY_MPS] +
129 u[TConfig::INPUT_DTHETA_RADPS] * x[TConfig::STATE_VZ_MPS]) /
130 tam::constants::g_earth;
131
132 reference_orientation_.orientation_rad.y =
133 std::asin(std::clamp(std::isnan(theta_value) ? 0.0 : theta_value, -0.5, 0.5));
134
135 return reference_orientation_;
136 };
