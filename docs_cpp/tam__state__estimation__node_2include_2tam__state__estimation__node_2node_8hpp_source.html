315
319 void model_update_callback(void)
320 {
321 // iterate through the odometry buffer and input the delay compensated odmetry measurements
322 const auto start_delay_comp = std::chrono::high_resolution_clock::now();
323 const double covariance_scale = state_estimation_node_params_.virtual_covariance_scale;
324 for (const auto& measurement : measurement_channels_) {
325 if (measurement.received) {
326 tam::types::control::Odometry input = measurement.odometry;
327 // compensate the message delay and use the measurement input if the message isn't to old
328 // clang-format off
329 const double time_diff = static_cast<double>((this->now() - measurement.measurement_stamp).nanoseconds()) / 1e9; // NOLINT
330 // clang-format on
331 if (delay_compensation_->compensate(input, time_diff)) {
332 // increase the covariance of the measurement due to the delay compensation
333 std::transform(measurement.odometry.pose_covariance.begin(), measurement.odometry.pose_covariance.end(),
334 input.pose_covariance.begin(),
335 [covariance_scale, time_diff](double x) { return x + x * covariance_scale * time_diff; });
336 // set state estimation localization input
337 if (measurement.config.position_m.active)
338 state_estimation_->set_input_position(input, measurement.config.position_m.num);
339 if (measurement.config.orientation_rad.active)
340 state_estimation_->set_input_orientation(input, measurement.config.orientation_rad.num,
341 measurement.config.orientation_rad.mask[0], measurement.config.orientation_rad.mask[1],
342 measurement.config.orientation_rad.mask[2]);
343 if (measurement.config.linear_velocity_mps.active)
344 state_estimation_->set_input_linear_velocity(input, measurement.config.linear_velocity_mps.num);
345 } else {
346 // set sensor state to error if the message is older than a second
347 if (measurement.config.position_m.active)
348 state_estimation_->set_input_position_status(
349 tam::types::ErrorLvl::ERROR, measurement.config.position_m.num);
350 if (measurement.config.orientation_rad.active)
351 state_estimation_->set_input_orientation_status(
352 tam::types::ErrorLvl::ERROR, measurement.config.orientation_rad.num);
353 if (measurement.config.linear_velocity_mps.active)
354 state_estimation_->set_input_linear_velocity_status(
355 tam::types::ErrorLvl::ERROR, measurement.config.linear_velocity_mps.num);
356 }
357 }
358 }
359 // runtime logging for the delay compensation
360 const auto end_delay_comp = std::chrono::high_resolution_clock::now();
361 const double duration_delay_comp_us = std::chrono::duration_cast<std::chrono::microseconds>(
362 end_delay_comp - start_delay_comp).count();
363
364 // initialize the initial position and heading of the state estimation
365 // set set_initial_state will return true if the state was set successfull
366 // we dont publish any message before the state estimation was initialized successfully
367 if (initialize_) {
368 if (state_estimation_->set_initial_state()) {
369 initialize_ = false;
370 RCLCPP_INFO(this->get_logger(), "Initial state of the state estimation has been set");
371 monitor_->initialization_finished();
372 }
373 monitor_->set_message(state_estimation_->get_state_machine_status_msg());
374 } else {
375 // the state estimation was initialized successfully
376 // step the state estimation
377 const auto start_step = std::chrono::high_resolution_clock::now();
378 state_estimation_->step();
379 const auto end_step = std::chrono::high_resolution_clock::now();
380 const double duration_step_us = std::chrono::duration_cast<std::chrono::microseconds>(
381 end_step - start_step).count();
382
383 // update the error level of the state estimatiion node
384 const auto start_publish = std::chrono::high_resolution_clock::now();
385
386 monitor_->set_error_lvl("internal state", state_estimation_->get_status());
387 monitor_->set_message(state_estimation_->get_state_machine_status_msg());
388
389 // get the state estimation output
390 odometry_output_ = state_estimation_->get_odometry();
391 acceleration_output_ = state_estimation_->get_acceleration();
392
393 // Publish estimated states with a fixed, non-trivially compensatable delay
394 // on the IMU and linear velocity measurements.
395 this->time_pub_ = get_clock()->now();
396 this->time_pub_ -= std::chrono::milliseconds(state_estimation_node_params_.average_input_delay_ms);
397
398 // publish state estimation output
399 publish_odometry(time_pub_);
400 publish_acceleration(time_pub_);
401
402 // publish dynamic transforms
403 publish_dynamic_transforms(time_pub_);
404
405 // runtime logging for the publisher
406 const auto end_publish = std::chrono::high_resolution_clock::now();
407 const double duration_publish_us = std::chrono::duration_cast<std::chrono::microseconds>(
408 end_publish - start_publish).count();
409
410 // Update the Debug Values of the state estimation class
411 const auto start_logging = std::chrono::high_resolution_clock::now();
412 state_estimation_->get_state_machine_debug_output(logger_);
413 state_estimation_->get_kalman_filter_debug_output(logger_);
414
415 // iterate through the odometry buffer and publish the debug information
416 for (const auto& measurement : measurement_channels_) log_measurement_input(measurement);
417 const auto end_logging = std::chrono::high_resolution_clock::now();
418 const double duration_logging_us = std::chrono::duration_cast<std::chrono::microseconds>(
419 end_logging - start_logging).count();
420 // time between callbacks
421 const rclcpp::Time callback_time = this->get_clock()->now();
422 const double callback_dt_us = static_cast<double>(
423 (callback_time - last_callback_time_).to_chrono<std::chrono::microseconds>().count()
424 );
425 last_callback_time_ = callback_time;
426
427 // runtime logging
428 logger_->log("runtime/delay_compensation", duration_delay_comp_us);
429 logger_->log("runtime/step", duration_step_us);
430 logger_->log("runtime/publish", duration_publish_us);
431 logger_->log("runtime/logging", duration_logging_us);
432 logger_->log("runtime/callback_dt", callback_dt_us);
433
434 // publish the debug message
435 tsl_publisher_.trigger();
436
437 // update the delay compensation buffer
438 delay_compensation_->update_vehicle_odometry(odometry_output_);
439 }

442 // publisher function
446 void publish_odometry(rclcpp::Time time_pub)
447 {
448 // convert odometry type to a nav_msgs::msg::Odometry
449 nav_msgs::msg::Odometry odometry_msg = tam::type_conversions::odometry_msg_from_type(odometry_output_);
450
451 // construct message header

530 void log_measurement_input(const tam::types::state::measurement& measurement)
531 {
532 if (!measurement.config.position_m.active || !measurement.config.orientation_rad.active) return;
533
534 // publish the delay compensated and transformed measurements as debug
535 std::string debug_namespace = "pose/" + measurement.config.name;
536 debug_namespace.reserve(debug_namespace.size() + 64);
537
538 // lamda to log the transformed odometry as debug
539 auto log_odometry = [&](const std::string& prefix, const tam::types::control::Odometry& odom, double delay) {
540 logger_->log(debug_namespace + prefix + "/position_m/x", odom.position_m.x);
541 logger_->log(debug_namespace + prefix + "/position_m/y", odom.position_m.y);
542 logger_->log(debug_namespace + prefix + "/position_m/z", odom.position_m.z);
543 logger_->log(debug_namespace + prefix + "/orientation_rad/roll", odom.orientation_rad.x);
544 logger_->log(debug_namespace + prefix + "/orientation_rad/pitch", odom.orientation_rad.y);
545 logger_->log(debug_namespace + prefix + "/orientation_rad/yaw", odom.orientation_rad.z);
546 logger_->log(debug_namespace + prefix + "/velocity_mps/x", odom.velocity_mps.x);
547 logger_->log(debug_namespace + prefix + "/velocity_mps/y", odom.velocity_mps.y);
548 logger_->log(debug_namespace + prefix + "/velocity_mps/z", odom.velocity_mps.z);
549 logger_->log(debug_namespace + prefix + "/msg_delay", delay);
550 };
551
552 if (measurement.received) {
553 // once delay compensated measurements
554 tam::types::control::Odometry delay_compensated_odometry = measurement.odometry;
555 const double measured_time_diff =
556 static_cast<double>((measurement.received_stamp - measurement.measurement_stamp).nanoseconds()) / 1e9;
557 delay_compensation_->compensate(delay_compensated_odometry, measured_time_diff);
558 log_odometry("/measured", delay_compensated_odometry, measured_time_diff);
559
560 // Continously forward predicted virtual measurements
561 tam::types::control::Odometry virtual_odometry = measurement.odometry;
562 const double virtual_time_diff =
563 static_cast<double>((this->now() - measurement.measurement_stamp).nanoseconds()) / 1e9;
564 delay_compensation_->compensate(virtual_odometry, virtual_time_diff);
565 log_odometry("/virtual", virtual_odometry, virtual_time_diff);
566 } else {
567 log_odometry("/measured", {}, 0.0);
568 log_odometry("/virtual", {}, 0.0);
569 }

572 // Subscription callbacks sensors
573 void measurement_callback(
574 const nav_msgs::msg::Odometry::ConstSharedPtr msg, tam::types::state::measurement& measurement)
575 {
576 // input position of the first sensor position
577 measurement.odometry = tam::type_conversions::odometry_type_from_msg(*msg);
578
579 // Transform pose to center of gravity
580 if (measurement.config.position_m.active || measurement.config.orientation_rad.active) {
581 try {
582 // Try to look up the transform until the first successful lookup
583 if (!measurement.static_translation_valid) {
584 geometry_msgs::msg::TransformStamped transform = tf_buffer_->lookupTransform(
585 msg->child_frame_id, state_estimation_node_params_.child_frame_id, tf2::TimePointZero);
586 measurement.static_translation = tf2::Vector3(transform.transform.translation.x,
587 transform.transform.translation.y, transform.transform.translation.z);
588 measurement.static_translation_valid = true;
589 }
590
591 // build the rotation matrix
592 tf2::Matrix3x3 rotation_matrix;
593 if (initialize_) {
594 rotation_matrix.setEulerYPR(measurement.odometry.orientation_rad.z, measurement.odometry.orientation_rad.y,
595 measurement.odometry.orientation_rad.x);
596 } else {
597 rotation_matrix.setEulerYPR(
598 odometry_output_.orientation_rad.z, odometry_output_.orientation_rad.y, odometry_output_.orientation_rad.x);
599 }
600
601 // rotate translation in the local cartesian frame
602 const tf2::Vector3 transformation = rotation_matrix * measurement.static_translation;
603
604 // transform from sensor frame to vehicle CoG
605 measurement.odometry.position_m = measurement.odometry.position_m +
606 tam::types::common::Vector3D<double>{transformation.x(), transformation.y(), transformation.z()};
607 } catch (tf2::TransformException& ex) {
608 RCLCPP_INFO(this->get_logger(), "Failed to transform: %s", ex.what());
609 return;
610 }
611 }
612 // buffer the received odometry and account for constant delay
613 rclcpp::Time measurement_timestamp = msg->header.stamp;
614
615 // Compute the additional delay
616 const auto additional_odom_delay_ms = std::chrono::milliseconds(measurement.config.additional_delay);
617 const auto average_input_delay_ms = std::chrono::milliseconds(state_estimation_node_params_.average_input_delay_ms);
618 if ((measurement_timestamp - additional_odom_delay_ms + average_input_delay_ms) <= this->now()) {
619 measurement_timestamp -= additional_odom_delay_ms - average_input_delay_ms;
620 } else {
621 measurement_timestamp -= additional_odom_delay_ms;
622 }
623
624 // set timestamps in buffer
625 measurement.measurement_stamp = measurement_timestamp;
626 measurement.received_stamp = this->now();
627 measurement.received = true;
628 }
