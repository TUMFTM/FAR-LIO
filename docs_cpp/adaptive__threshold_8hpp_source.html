55 void set_model_deviation(
56 [[maybe_unused]] const Sophus::SE3f& init_guess, [[maybe_unused]] const Sophus::SE3f& pose_registered) override
57 {
58 if (!this->initialized_) {
59 // If not initialized, just initialize the module and return without updating the model error
60 this->previous_init_guess_ = init_guess;
61 this->initialized_ = true;
62 return;
63 }
64 // Compute deviation between initial guess and registration
65 const Sophus::SE3f model_deviation = init_guess.inverse() * pose_registered;
66 // Compute model error
67 const double model_error = [&]() {
68 const double theta = Eigen::AngleAxisf(model_deviation.rotationMatrix()).angle();
69 const double delta_rot = 2.0 * this->config_.max_correspondence_range * std::sin(theta / 2.0);
70 const double delta_trans = model_deviation.translation().norm();
71 return delta_trans + delta_rot;
72 }();
73 // Update the model error given the vehicle is actually moving translationally
74 // and the model error is significant
75 if (this->initialized_ &&
76 (this->previous_init_guess_.inverse() * init_guess).translation().norm() >
77 3 * this->config_.min_motion_threshold &&
78 model_error > this->config_.min_motion_threshold) {
79 this->model_sse_ += model_error * model_error;
80 this->num_samples_++;
81 this->debug_.model_error = model_error;
82 }
83 }
84
88 double get_threshold() override
89 {
90 this->debug_.internal_threshold = std::sqrt(this->model_sse_ / this->num_samples_);
91 this->debug_.current_threshold = std::min(this->debug_.internal_threshold, 10.0);
92 return this->debug_.current_threshold;
93 };
94
98 void init() override { model_sse_ = this->config_.initial_threshold * this->config_.initial_threshold; }
