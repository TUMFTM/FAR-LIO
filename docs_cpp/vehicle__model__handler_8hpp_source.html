56 void input_wheel_angular_velocities([[maybe_unused]] const tam::types::common::DataPerWheel<double>& wheel)
57 {
58 // calculate the average velocity over all wheels and apply the wheelspeed scale
59 // if a vehicle model is configured
60 if constexpr (TModel != VehicleModel::Kinematic) {
61 dyn_tire_radii_scale_ = tam::helpers::numerical::interp(vehicle_odometry_.velocity_mps.x,
62 vehicle_model_params_.rr_vel_scale_vel_points_mps, vehicle_model_params_.rr_vel_scale_scale_factors);
63
64 const double average_absolute_velocity =
65 ((wheel.front_left + wheel.front_right) * vehicle_model_params_.tyre_radius_front_m * dyn_tire_radii_scale_ +
66 (wheel.rear_left + wheel.rear_right) * vehicle_model_params_.tyre_radius_rear_m * dyn_tire_radii_scale_) /
67 4;
68
69 // velocity in x- and y-direction calculated from the linear STM
70 if constexpr (TModel == VehicleModel::SingleTrack) {
71 vehicle_odometry_.velocity_mps.x = average_absolute_velocity * cos(side_slip_angle_);
72 vehicle_odometry_.velocity_mps.y = average_absolute_velocity * sin(side_slip_angle_);
73 vehicle_odometry_.velocity_mps.z = 0.0;
74 } else {
75 // average velocity over all wheels in x-direction
76 vehicle_odometry_.velocity_mps.x = average_absolute_velocity;
77 vehicle_odometry_.velocity_mps.y = 0.0;
78 vehicle_odometry_.velocity_mps.z = 0.0;
79 }
80 }
81 };

97 void input_steering_angle([[maybe_unused]] double steering_angle)
98 {
99 // update the side slip angle using a linear STM
100 if constexpr (TModel == VehicleModel::SingleTrack) {
101 side_slip_angle_ = std::atan(std::tan(steering_angle) * vehicle_model_params_.l_WheelbaseR_m /
102 (vehicle_model_params_.l_WheelbaseF_m + vehicle_model_params_.l_WheelbaseR_m));
103 }
104 };

127 tam::types::ErrorLvl get_status(void) const
128 {
129 // only return OK if the output of the vehicle model should be fused
130 // and all needed sensors are OK
131 if constexpr (TModel != VehicleModel::Kinematic) {
132 if (wheelspeed_status_ != tam::types::ErrorLvl::OK) return tam::types::ErrorLvl::ERROR;
133 if constexpr (TModel == VehicleModel::SingleTrack) {
134 if (steering_angle_status_ != tam::types::ErrorLvl::OK) return tam::types::ErrorLvl::ERROR;
135 }
136
137 return tam::types::ErrorLvl::OK;
138 }
139
140 // else return an error such that the solution is not fused
141 return tam::types::ErrorLvl::ERROR;
142 };
