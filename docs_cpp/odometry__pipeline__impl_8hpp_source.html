79template <typename TConfig>
80types::Odometry OdometryPipeline<TConfig>::register_frame(
81 const std::vector<types::Point<TConfig>>& frame, const std::uint64_t frame_stamp)
82{
83 // If an async map update staged a replacement, use it
84 this->map_->consume_map_switch();
85
86 // Get initial guess
87 types::PoseStamped init_guess = this->model_->get_initial_guess(frame_stamp);
88 // Take z-position from buffered pose if it was valid
89 if (this->odometry_.status.level == types::DiagnosticLevel::OK && this->config_.buffer_z) {
90 init_guess.pose.translation().z() = this->odometry_.pose.pose.translation().z();
91 }
92
93 // Catch empty frame
94 // Copying an empty vector to the GPU leads to an invalid device ordinal error
95 if (frame.empty()) {
96 types::Odometry odom{};
97 odom.stamp = frame_stamp;
98 odom.pose.pose = init_guess.pose;
99 odom.status.level = types::DiagnosticLevel::ERROR;
100 odom.status.message = "empty cloud";
101 return odom;
102 }
103
104 // Preprocessing
105 const double map_resolution = this->map_->get_resolution();
106
107 // Allocate frame registration and map
108#ifdef __CUDACC__
109 nvtxRangePush("setup_frame");
110 // Load the data to the GPU and double downsample frame
111 frame_.resize(frame.size());
112 cudaMemcpy(thrust::raw_pointer_cast(frame_.data()), frame.data(), frame.size() * sizeof(types::Point<TConfig>),
113 cudaMemcpyHostToDevice);
114 // Undistort frame if configured
115 if (this->config_.undistort) {
116 // Perform in-place undistortion
117 this->distortion_->undistort(frame_, frame_stamp, this->stream_);
118 }
119 // Preprocess frame if configured
120 if (this->config_.preprocess) {
121 if (!this->preprocessing_->preprocess(frame_, this->stream_)) {
122 types::Odometry odom{};
123 odom.stamp = frame_stamp;
124 odom.pose.pose = init_guess.pose;
125 odom.status.level = types::DiagnosticLevel::ERROR;
126 odom.status.message = "Empty cloud after preprocessing!";
127 return odom;
128 }
129 }
130 // Start timer for downsample time
131 auto start = std::chrono::high_resolution_clock::now();
132 // Downsample frame
133 if (this->config_.downsample) {
134 cuda::voxel_doubledownsample_device(
135 frame_, frame_map_, frame_registration_, this->map_->get_config().voxel_size, 50000, this->stream_);
136 } else {
137 // Downsample the point cloud to match the map resulution to ensure a faster point insertion
138 // This is necessary restict the number of threads that want to insert into the same voxel
139 cuda::voxel_downsample(frame_, frame_map_, map_resolution, 50000, this->stream_);
140 frame_registration_.resize(frame_map_.size());
141 thrust::copy(
142 thrust::cuda::par.on(this->stream_), frame_map_.begin(), frame_map_.end(), frame_registration_.begin());
143 }
144 nvtxRangePop();
145#else
146 std::vector<types::Point<TConfig>> frame_preprocessed = frame;
147 // Undistort frame if configured
148 if (this->config_.undistort) {
149 this->distortion_->undistort(frame_preprocessed, frame_stamp);
150 }
151 if (this->config_.preprocess) {
152 if (!this->preprocessing_->preprocess(frame_preprocessed)) {
153 types::Odometry odom{};
154 odom.stamp = frame_stamp;
155 odom.pose.pose = init_guess.pose;
156 odom.status.level = types::DiagnosticLevel::ERROR;
157 odom.status.message = "Empty cloud after preprocessing!";
158 return odom;
159 }
160 }
161 // Start timer for downsample
162 auto start = std::chrono::high_resolution_clock::now();
163 // Downsample frame
164 if (this->config_.downsample) {
165 std::tie(frame_registration_, frame_map_) =
166 voxel_doubledownsample(frame_preprocessed, this->map_->get_config().voxel_size);
167 } else {
168 // Downsample the point cloud to match the map resulution to ensure a faster point insertion
169 frame_registration_ = voxel_downsample(frame_preprocessed, map_resolution);
170 frame_map_ = frame_registration_;
171 }
172#endif
173
174 // Update debug values for downsampling time
175 this->debug_.num_points_frame = frame.size();
176 // clang-format off
177 this->debug_.downsample_time = std::chrono::duration_cast<std::chrono::microseconds>(std::chrono::high_resolution_clock::now() - start).count() * 1.0e-3; // NOLINT
178 // clang-format on
179
180 // Get correspondence threshold
181 const double sigma = this->threshold_->get_threshold();
182
183 // Register frame
184#ifdef __CUDACC__
185 const Sophus::SE3f pose_registered = this->registration_->register_frame(
186 frame_registration_, this->map_.get(), init_guess.pose, 3.0 * sigma, sigma / 3.0, this->stream_);
187#else
188 const Sophus::SE3f pose_registered = this->registration_->register_frame(
189 frame_registration_, this->map_.get(), init_guess.pose, 3.0 * sigma, sigma / 3.0);
190#endif
191 // Create a Pose with timestamp from the registered pose
192 types::PoseStamped pose_registered_stamped;
193 pose_registered_stamped.stamp = frame_stamp;
194 pose_registered_stamped.pose = pose_registered;
195
196 // Compute the tangent (velocity) using the velocity handler
197 types::TangentStamped tangent{};
198 tangent = this->velocity_->get_tangent({}, pose_registered_stamped);
199
200 // Get diagnostic status
201 const types::DiagnosticStatus diag_status = this->diagnostic_->get_diagnostic_status(
202 pose_registered_stamped, tangent, init_guess, this->registration_->get_registration_status());
203
204 // Set model deviation
205 this->threshold_->set_model_deviation(init_guess.pose, pose_registered);
206 // Set pose to model handler to compute initial guess for next frame if initial guess
207 // is not captured from EXTERNALGUESS
208 if constexpr (TConfig::MODEL != types::ModelType::EXTERNALGUESS) {
209 this->model_->set_pose(pose_registered_stamped, true);
210 }
211 // Update map if configured and registration was successful
212 if (this->config_.update_map) {
213 std::variant<int16_t, types::AdaptiveMapDensity> map_density = TConfig::MAX_POINTS_PER_VOXEL;
214#ifdef __CUDACC__
215 if constexpr (types::FRAMEMAP<TConfig>) {
216 const types::AdaptiveMapDensity adaptive_density =
217 this->registration_->get_frame_map_density(20, pose_registered.translation(), this->stream_);
218 if (adaptive_density.valid) map_density = adaptive_density;
219 }
220 this->map_->update_points(frame_map_, pose_registered, map_density, 1, TConfig::NUM_NEIGHBORS, this->stream_);
221#else
222 if constexpr (types::FRAMEMAP<TConfig>) {
223 const types::AdaptiveMapDensity adaptive_density =
224 this->registration_->get_frame_map_density(20, pose_registered.translation());
225 if (adaptive_density.valid) map_density = adaptive_density;
226 }
227 this->map_->update_points(frame_map_, pose_registered, map_density, 1, TConfig::NUM_NEIGHBORS);
228#endif
229 }
230
231 // Build Odometry output
232 types::Odometry odom;
233 odom.stamp = frame_stamp;
234 odom.pose.pose = pose_registered;
235 odom.tangent.tangent = tangent.tangent;
236 odom.status = diag_status;
237
238 // Get correspondences of final registration result for covariance estimation
239 std::vector<types::Correspondence<TConfig>> correspondences{};
240 if constexpr (TConfig::COVARIANCE != types::CovarianceType::CONSTANT) {
241#ifdef __CUDACC__
242 const auto& correspondences = this->registration_->get_correspondences(
243 frame_registration_, this->map_.get(), pose_registered, 3.0 * sigma, this->stream_);
244 // Sync stream
245 cudaStreamSynchronize(this->stream_);
246#else
247 const auto& correspondences =
248 this->registration_->get_correspondences(frame_registration_, this->map_.get(), pose_registered, 3.0 * sigma);
249#endif
250 }
251 // Get covariances
252 odom.pose.covariance = this->covariance_->get_pose_covariance(pose_registered, correspondences, sigma / 3.0);
253 odom.tangent.covariance = this->covariance_->get_tangent_covariance();
254
255 // Buffer odometry
256 this->odometry_ = odom;
257
258#ifdef __CUDACC__
259 // Copy downsampled frame to host when in debug mode
260 if (this->config_.debug_mode) {
261 this->frame_downsampled_.resize(frame_registration_.size());
262 thrust::copy(frame_registration_.begin(), frame_registration_.end(), this->frame_downsampled_.begin());
263 }
264#else
265 this->frame_downsampled_ = frame_registration_;
266#endif
267 // Log the whole pipeline time
268 // clang-format off
269 this->debug_.pipeline_time = std::chrono::duration_cast<std::chrono::microseconds>(std::chrono::high_resolution_clock::now() - start).count() * 1.0e-3; // NOLINT
270 // clang-format on
271 return odom;
272}

278template <typename TConfig>
279void OdometryPipeline<TConfig>::init()
280{
281 this->distortion_->init(this->config_.num_threads);
282 this->map_->init(this->config_.num_threads, this->config_.update_map);
283 this->registration_->init(this->config_.num_threads);
284 this->preprocessing_->init();
285 this->threshold_->init();
286 if constexpr (types::FRAMEMAP<TConfig>) {
287 this->registration_->init_frame_map(this->map_->get_config());
288 }

602template <typename TConfig>
603void OdometryPipeline<TConfig>::set_config(tam::pmg::ParamReferenceManager* pmg)
604{
605 // clang-format off
606 pmg->declare_parameter("pipeline.update_map", &this->config_.update_map, false, tam::pmg::ParameterType::BOOL, "Update map with new points"); // NOLINT
607 pmg->declare_parameter("pipeline.undistort", &this->config_.undistort, true, tam::pmg::ParameterType::BOOL, "Apply motion distortion correction"); // NOLINT
608 pmg->declare_parameter("pipeline.preprocess", &this->config_.preprocess, true, tam::pmg::ParameterType::BOOL, "Preprocess point cloud"); // NOLINT
609 pmg->declare_parameter("pipeline.downsample", &this->config_.downsample, true, tam::pmg::ParameterType::BOOL, "Downsample point cloud"); // NOLINT
