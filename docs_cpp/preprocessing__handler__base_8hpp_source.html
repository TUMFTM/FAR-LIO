42void filter_points(thrust::device_vector<types::Point<TConfig>>& points, Predicate predicate, const std::string& name,
43 std::unordered_map<std::string, std::int64_t>& num_points, std::unordered_map<std::string, double>& preprocess_time,
44 ::cuda::stream_ref stream = {})
45{
46 auto start = std::chrono::high_resolution_clock::now();
47 auto end = thrust::remove_if(thrust::cuda::par.on(stream.get()), points.begin(), points.end(), predicate);
48 points.resize(end - points.begin());
49
50 // Compute time
