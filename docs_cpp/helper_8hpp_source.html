44template <class TConfig>
45void parse_sensor_configs(rclcpp::Node* node, tam::pmg::MgmtInterface* param_manager_raw,
46 std::vector<tam::types::state::measurement>& measurement_channels,
47 std::vector<tam::types::state::input>& input_channels,
48 std::vector<std::pair<tam::types::state::measurements::identifier, std::string>>& identifier_names)
49{
50 std::unordered_map<std::string, std::map<std::string, rclcpp::ParameterValue>> measurement_map, input_map;
51 const auto overrides = node->get_node_parameters_interface()->get_parameter_overrides();
52 for (const auto& [name, value] : overrides) {
53 std::vector<std::string> tokens;
54 boost::split(tokens, name, boost::is_any_of("."));
55 if (tokens.size() >= 3 && tokens[0] == "measurements") {
56 const std::string key = boost::join(boost::make_iterator_range(tokens.begin() + 2, tokens.end()), ".");
57 measurement_map[tokens[1]][key] = value;
58 node->declare_parameter(name, value);
59 } else if (tokens.size() >= 4 && tokens[0] == "inputs" && tokens[1] == "imus") {
60 const std::string key = boost::join(boost::make_iterator_range(tokens.begin() + 3, tokens.end()), ".");
61 input_map[tokens[2]][key] = value;
62 node->declare_parameter(name, value);
63 }
64 }
65
66 // Match ROS Overrides to Measurement Configuration Data Structure
67 uint8_t num_position{}, num_orientation{}, num_linear_velocity{};
68 for (const auto& [name, parameter] : measurement_map) {
69 tam::types::state::measurement channel{};
70 tam::types::state::config::measurement& cfg = channel.config;
71 cfg.name = name;
72
73 // Handling of false subscription config
74 if (!parameter.count("message.topic"))
75 throw std::invalid_argument(
76 "[StateEstimationCPP]: Measurement input configuration for " + cfg.name + " does not provide a topic");
77 if (!parameter.count("message.type"))
78 throw std::invalid_argument(
79 "[StateEstimationCPP]: Measurement input configuration for " + cfg.name + " does not provide a topic type");
80
81 // Match Subscription config to data structure
82 // clang-format off
83 cfg.measurement_sub.topic = parameter.at("message.topic").template get<std::string>();
84 cfg.measurement_sub.type = parameter.at("message.type").template get<std::string>();
85 cfg.measurement_sub.timeout_ms = parameter.count("message.timeout_ms") ? parameter.at("message.timeout_ms").template get<int32_t>() : 500; // NOLINT
86 cfg.measurement_sub.active = true;
87 // clang-format on
88
89 // Only nav_msgs/msg/Odometry measurement topics are currently supported
90 if (cfg.measurement_sub.type != "nav_msgs/msg/Odometry")
91 throw std::invalid_argument(
92 "[StateEstimationCPP]: Measurement input configuration for " + cfg.name + " has an unimplemented topic type '"
93 + cfg.measurement_sub.type + "', only 'nav_msgs/msg/Odometry' is supported");
94
95 // Handling of status subscription config
96 // clang-format off
97 if (parameter.count("status.topic")) {
98 cfg.status_sub.topic = parameter.at("status.topic").template get<std::string>();
99 cfg.status_sub.timeout_ms = parameter.count("status.timeout_ms") ? parameter.at("status.timeout_ms").template get<int32_t>() : 500; // NOLINT
100 cfg.status_sub.active = true;
101 } else {
102 std::cout << "[StateEstimationCPP]: Measurement input configuration for " + cfg.name + " does not provide a status topic, input is used without status" << std::endl; // NOLINT
103 cfg.status_sub.active = false;
104 }
105
106 // Match additional delay on the topic
107 cfg.additional_delay = parameter.count("message.additional_delay_ms") ? parameter.at("message.additional_delay_ms").template get<int32_t>() : 0; // NOLINT
108
109 // Lamda to match fusion configs to the sensor
110 auto setup_fusion_config = [&](const std::string& prefix, tam::types::state::config::fusion& fusion, uint8_t& num) { // NOLINT
111 if (!parameter.count(prefix + ".R_init") || !parameter.count(prefix + ".outlier_bounds")) return;
112 fusion.R_init = parameter.at(prefix + ".R_init").template get<std::vector<double>>();
113 fusion.outlier_bounds = parameter.at(prefix + ".outlier_bounds").template get<std::vector<double>>();
114 // a measurement-noise variance must be finite and strictly positive, otherwise the innovation
115 // covariance S = H P H^T + R can become singular and S.inverse() produces NaNs in the update
116 for (const double r : fusion.R_init)
117 if (!std::isfinite(r) || r <= 0.0)
118 throw std::invalid_argument("[StateEstimationCPP]: Measurement input configuration for " + cfg.name + " has a non-positive R_init entry for " + prefix); // NOLINT
119 fusion.mask = parameter.count(prefix + ".mask") ? parameter.at(prefix + ".mask").template get<std::vector<bool>>() : std::vector<bool>(fusion.R_init.size(), true); // NOLINT
120 fusion.num = num++;
121 fusion.active = true;
122 if (fusion.R_init.size() != fusion.outlier_bounds.size() || fusion.R_init.size() != fusion.mask.size())
123 throw std::invalid_argument("[StateEstimationCPP]: Measurement input configuration for " + cfg.name + " does not provid consistent sizes for " + prefix); // NOLINT
124 };
125
126 // Match fusion configs for the measurements
127 setup_fusion_config("position_m", cfg.position_m, num_position);
128 setup_fusion_config("orientation_rad", cfg.orientation_rad, num_orientation);
129 setup_fusion_config("linear_velocity_mps", cfg.linear_velocity_mps, num_linear_velocity);
130 measurement_channels.push_back(channel);
131 // clang-format on
132 }
133
134 // Match ROS Overrides to Input Configuration Data Structure
135 uint8_t num_imu{};
136 for (const auto& [name, parameter] : input_map) {
137 tam::types::state::input channel{};
138 tam::types::state::config::input& cfg = channel.config;
139 cfg.name = name;
140
141 // Handling of false subscription config
142 if (!parameter.count("message.topic"))
143 throw std::invalid_argument(
144 "[StateEstimationCPP]: Input configuration for " + cfg.name + " does not provide a topic");
145 if (!parameter.count("message.type"))
146 throw std::invalid_argument(
147 "[StateEstimationCPP]: Input configuration for " + cfg.name + " does not provide a topic type");
148
149 // Match Subscription config to data structure
150 // clang-format off
151 cfg.measurement_sub.topic = parameter.at("message.topic").template get<std::string>();
152 cfg.measurement_sub.type = parameter.at("message.type").template get<std::string>();
153 cfg.measurement_sub.timeout_ms = parameter.count("message.timeout_ms") ? parameter.at("message.timeout_ms").template get<int32_t>() : 500; // NOLINT
154 cfg.measurement_sub.active = true;
155
156 // Only sensor_msgs/msg/Imu input topics are currently supported
157 if (cfg.measurement_sub.type != "sensor_msgs/msg/Imu")
158 throw std::invalid_argument(
159 "[StateEstimationCPP]: Input configuration for " + cfg.name + " has an unimplemented topic type '"
160 + cfg.measurement_sub.type + "', only 'sensor_msgs/msg/Imu' is supported");
161
162 // Match IMU filter and backup settings
163 cfg.filter_coefficients = parameter.count("filter_coefficients") ? parameter.at("filter_coefficients").template get<std::vector<double>>() : std::vector<double>({1.0}); // NOLINT
164 cfg.backup = parameter.count("backup") ? parameter.at("backup").template get<bool>() : false;
165 cfg.num = !cfg.backup ? num_imu++ : TConfig::NUM_IMU_MEASUREMENT + TConfig::NUM_BACKUP_IMU_MEASUREMENT - 1;
166 identifier_names.emplace_back(tam::types::state::measurements::identifier{tam::types::state::measurements::IMU, cfg.num}, cfg.name); // NOLINT
167 input_channels.push_back(channel);
168
169 // Set IMU Filter Coefficients in the IMU handler
170 param_manager_raw->set_value("imu_" + std::to_string(cfg.num + 1) + "_filter_coefficients", cfg.filter_coefficients); // NOLINT
171 // clang-format on
172 }
173}
