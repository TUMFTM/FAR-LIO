103 types::Correspondence<TConfig> search_closest_neighbor(
104 const types::Point<TConfig>& point, const int16_t adjacent_voxels = 1) const override
105 {
106 // Convert the point to voxel coordinates
107 const auto& voxel = point_to_voxel<TConfig>(point, this->config_.voxel_size);
108 // Get nearby voxels on the map
109 const auto& neighbor_voxels = get_adjacent_voxels(voxel, adjacent_voxels);
110 // Iterate over the neighbor voxels
111 std::vector<types::Point<TConfig>> neighbors{};
112 int16_t num_neighbors = (2 * adjacent_voxels + 1) * (2 * adjacent_voxels + 1) * (2 * adjacent_voxels + 1);
113 neighbors.reserve(num_neighbors * TConfig::MAX_POINTS_PER_VOXEL);
114 std::for_each(neighbor_voxels.cbegin(), neighbor_voxels.cend(), [&](const auto& neighbor) {
115 auto search = this->get_active_map().find(neighbor);
116 if (search != this->get_active_map().end()) {
117 const auto& map_points = search.value();
118 if (!map_points.empty()) {
119 for (const auto& map_point : map_points) {
120 neighbors.emplace_back(map_point);
121 }
122 }
123 }
124 });
125 // Find the closest neighbor
126 types::Correspondence<TConfig> correspondence{};
127 correspondence.frame = point;
128 correspondence.distance = std::numeric_limits<double>::max();
129 if (!neighbors.empty()) {
130 std::for_each(neighbors.cbegin(), neighbors.cend(), [&](const auto& neighbor) {
131 float distance = (neighbor.pos - point.pos).squaredNorm();
132 if (distance < correspondence.distance) {
133 correspondence.map = neighbor;
134 correspondence.distance = distance;
135 }
136 });
137 }
138 return correspondence;
139 }

152 void add_points(const std::vector<types::Point<types::Point_XYZ>>& points,
153 const std::variant<int16_t, types::AdaptiveMapDensity>& map_density = TConfig::MAX_POINTS_PER_VOXEL,
154 const int16_t adjacent_voxels = 1, const int16_t num_neighbors = TConfig::NUM_NEIGHBORS,
155 const bool use_active_map = true) override
156 {
157 // Get reference to the target map and the neighbors vector
158 MapType& target_map = use_active_map ? this->get_active_map() : this->get_inactive_map();
159 tbb::concurrent_vector<types::Neighbors<TConfig>>& target_neighbors =
160 use_active_map ? this->get_active_neighbors() : this->get_inactive_neighbors();
161
162 // Clear map if frame map is enabled or its the inactive map
163 if (this->config_.frame_map || !use_active_map) {
164 target_map.clear();
165 }
166
167 // Reserve space for occupied voxel caching and neighbors
168 target_neighbors.clear();
169 target_neighbors.reserve(points.size());
170 const double map_resolution = this->get_resolution();
171 unsigned int counter = 0;
172 std::for_each(points.cbegin(), points.cend(), [&](const auto& point) {
173 const auto voxel = point_to_voxel<types::Point_XYZ>(point, this->config_.voxel_size);
174 auto search = target_map.find(voxel);
175 if (search != target_map.end()) {
176 std::vector<types::Point<TConfig>>& voxel_points = search.value();
177 // Compute max number of points within this voxel
178 int16_t max_points_per_voxel = TConfig::MAX_POINTS_PER_VOXEL;
179 if (std::holds_alternative<types::AdaptiveMapDensity>(map_density)) {
180 // clang-format off
181 const auto & adaptive_map_density = std::get<types::AdaptiveMapDensity>(map_density);
182 const float distance2origin = (point.pos - adaptive_map_density.origin).norm();
183 if (distance2origin < adaptive_map_density.range) {
184 max_points_per_voxel = std::max(adaptive_map_density.min_points, adaptive_map_density.max_points); // NOLINT
185 } else {
186 max_points_per_voxel = std::max(
187 adaptive_map_density.min_points,
188 static_cast<int16_t>(adaptive_map_density.max_points + (distance2origin - adaptive_map_density.range) *adaptive_map_density.max_points_scale)); // NOLINT
189 }
190 // clang-format on
191 } else {
192 max_points_per_voxel = std::get<int16_t>(map_density);
193 }
194 if (voxel_points.size() >= static_cast<size_t>(max_points_per_voxel) ||
195 std::any_of(voxel_points.cbegin(), voxel_points.cend(),
196 [&](const auto& voxel_point) { return (voxel_point.pos - point.pos).norm() < map_resolution; })) {
197 // Voxel is already full
198 return;
199 }
200 voxel_points.emplace_back(utils::convert_point<types::Point_XYZ, TConfig>(point));
201 counter++;
202 // Store pointer to point in tbb_neighbors_
203 types::Neighbors<TConfig> neighbors{};
204 neighbors.point = &voxel_points.back();
205 target_neighbors.emplace_back(neighbors);
206 } else {
207 std::vector<types::Point<TConfig>> voxel_points;
208 // Always reserve for maximum amount of points (compile time param)
209 voxel_points.reserve(TConfig::MAX_POINTS_PER_VOXEL);
210 voxel_points.emplace_back(utils::convert_point<types::Point_XYZ, TConfig>(point));
211 target_map.insert({voxel, std::move(voxel_points)});
212 counter++;
213 // Store pointer to point in tbb_neighbors_
214 types::Neighbors<TConfig> neighbors{};
215 neighbors.point = &target_map[voxel].back();
216 target_neighbors.emplace_back(neighbors);
217 }
218 });
219 // Set normals and covariances to points (only if enabled)
220 if constexpr (types::HASNORMALCOV<TConfig>) {
221 // Set normals and covariances to points
222 this->compute_normal_covariance(adjacent_voxels, std::min(num_neighbors, TConfig::NUM_NEIGHBORS), use_active_map);
223 }
224 // Update debug information (maintained only for active map)
225 if (!use_active_map) return;
226 this->debug_.points_added = counter;
227 this->debug_.voxel_added = this->get_active_map().size() - this->debug_.num_voxel;
228 this->debug_.num_voxel = this->get_active_map().size();
229 this->debug_.num_points = this->num_points();
230 }

241 void update_points(const std::vector<types::Point<TConfig>>& points, const Sophus::SE3f& pose,
242 const std::variant<int16_t, types::AdaptiveMapDensity>& map_density = TConfig::MAX_POINTS_PER_VOXEL,
243 const int16_t adjacent_voxels = 1, const int16_t num_neighbors = TConfig::NUM_NEIGHBORS) override
244 {
245 // Start timer
246 auto start = std::chrono::high_resolution_clock::now();
247
248 // Transform points to map frame and map point type
249 std::vector<types::Point<types::Point_XYZ>> points_transformed(points.size());
250 std::transform(points.cbegin(), points.cend(), points_transformed.begin(),
251 [&](const auto& point) { return utils::convert_point<TConfig, types::Point_XYZ>(point); });
252 utils::transform_points(pose, points_transformed);
253 const Eigen::Vector3f& origin = pose.translation();
254
255 // Add new points to the map
256 this->add_points(points_transformed, map_density, adjacent_voxels, num_neighbors);
257
258 // Recompute neighbors if normals and covariances are enabled
259 // NOTE: needs to be done before removal, as otherwise the pointers
260 // become invalid
261 if constexpr (types::HASNORMALCOV<TConfig>) {
262 // Set normals and covariances to points
263 this->compute_normal_covariance(adjacent_voxels, std::min(num_neighbors, TConfig::NUM_NEIGHBORS));
264 }
265
266 // Remove far points from the map
267 this->remove_far_points(origin);
268
269 // Compute elapsed time
270 // clang-format off
271 double time = std::chrono::duration_cast<std::chrono::microseconds>(std::chrono::high_resolution_clock::now() - start).count() * 1.0e-3; // NOLINT
272 // clang-format on
273 // Update debug information
274 this->debug_.update_time = time;
275 }

282 types::AdaptiveMapDensity get_density(double range)
283 {
284 double range2 = range * range;
285 unsigned int voxels = 0;
286 unsigned int points = 0;
287 for (auto it = this->get_active_map().cbegin(); it != this->get_active_map().cend();) {
288 const auto& [voxel, voxel_points] = *it;
289 const auto& pt = voxel_points.front();
290 if (pt.pos.squaredNorm() < range2) {
291 ++voxels;
292 points += voxel_points.size();
293 ++it;
294 } else {
295 ++it;
296 }
297 }
298 // clang-format off
299 types::AdaptiveMapDensity density{};
300 density.range = range;
301 density.max_points = (voxels > 0) ? static_cast<std::int16_t>(points / voxels) : 0; // NOLINT
302 density.min_points = TConfig::NUM_NEIGHBORS;
303 density.max_points_scale = -0.03f;
304 density.valid = density.max_points > 0;
305 // clang-format on
306 return density;
307 }

373 void search_closest_neighbors(types::Neighbors<TConfig>& point, const int16_t adjacent_voxels,
374 const int16_t num_neighbors, const bool use_active_map = true) const override
375 {
376 const MapType& target_map = use_active_map ? this->get_active_map() : this->get_inactive_map();
377 // Convert the point to voxel coordinates
378 const auto& voxel = point_to_voxel<TConfig>(*(point.point), this->config_.voxel_size);
379 // Get nearby voxels on the map
380 const auto& neighbor_voxels = get_adjacent_voxels(voxel, adjacent_voxels);
381 // Iterate over the neighbor voxels
382 std::vector<types::Point<TConfig>*> neighbors{};
383 int16_t num_neighbor_voxels = (2 * adjacent_voxels + 1) * (2 * adjacent_voxels + 1) * (2 * adjacent_voxels + 1);
384 neighbors.reserve(num_neighbor_voxels * TConfig::MAX_POINTS_PER_VOXEL);
385 std::for_each(neighbor_voxels.cbegin(), neighbor_voxels.cend(), [&](const auto& neighbor) {
386 auto search = target_map.find(neighbor);
387 if (search != target_map.end()) {
388 const auto& map_points = search.value();
389 if (!map_points.empty()) {
390 for (const auto& map_point : map_points) {
391 neighbors.emplace_back(const_cast<types::Point<TConfig>*>(&map_point));
392 }
393 }
394 }
395 });
396 if (!neighbors.empty()) {
397 std::for_each(neighbors.cbegin(), neighbors.cend(), [&](const auto& neighbor) {
398 float distance = (neighbor->pos - point.point->pos).norm();
399 if (point.num_neighbors == num_neighbors) {
400 // If we already have enough neighbors, check if the current one is closer
401 float max_distance = 0.0f;
402 int16_t largest_neighbor_index = -1;
403 // iterate through all existing neighbors
404 for (int16_t neighbor_idx = 0; neighbor_idx < num_neighbors; ++neighbor_idx) {
405 // if current distance is larger than maximal_distance and distance
406 // update
407 if (max_distance < point.distance[neighbor_idx] && distance < point.distance[neighbor_idx]) {
408 max_distance = point.distance[neighbor_idx];
409 largest_neighbor_index = neighbor_idx;
410 }
411 }
412 // replace largest neighbor with current one
413 if (largest_neighbor_index != -1) {
414 point.neighbor[largest_neighbor_index] = neighbor;
415 point.distance[largest_neighbor_index] = distance;
416 }
417 } else {
418 // If we have space for more neighbors, just add it
419 point.neighbor[point.num_neighbors] = neighbor;
420 point.distance[point.num_neighbors++] = distance;
421 }
422 });
423 }
424 return;
425 }

435 void compute_normal_covariance(
436 const int16_t adjacent_voxels, const int16_t num_neighbors, const bool use_active_map = true)
437 requires types::HASNORMALCOV<TConfig>
438 {
439 tbb::concurrent_vector<types::Neighbors<TConfig>>& target_neighbors =
440 use_active_map ? this->get_active_neighbors() : this->get_inactive_neighbors();
441 // Start timer
442 auto start = std::chrono::high_resolution_clock::now();
443
444 // Compute normals and covariances
445 using neighbor_iterator = tbb::concurrent_vector<types::Neighbors<TConfig>>::iterator;
446 tbb::parallel_for(tbb::blocked_range<neighbor_iterator>(target_neighbors.begin(), target_neighbors.end()),
447 [&](const tbb::blocked_range<neighbor_iterator>& r) {
448 std::for_each(r.begin(), r.end(), [&](auto& neighbor) {
449 // Search for correspondences (= closest neighbors)
450 this->search_closest_neighbors(neighbor, adjacent_voxels, num_neighbors, use_active_map);
451 // Set normal and covariance to frame point (= map point)
452 this->set_normal_covariance(neighbor, num_neighbors);
453 });
454 });
455 // Compute elapsed time
456 // clang-format off
457 double time = std::chrono::duration_cast<std::chrono::microseconds>(std::chrono::high_resolution_clock::now() - start).count() * 1.0e-3; // NOLINT
458 // clang-format on
459 // Update debug information
460 if (use_active_map) this->debug_.normal_cov_time = time;
461 }

467 void remove_far_points(const Eigen::Vector3f& origin)
468 {
469 const double max_distance2 = this->config_.max_distance * this->config_.max_distance;
470 unsigned int counter = 0;
471 for (auto it = this->get_active_map().begin(); it != this->get_active_map().end();) {
472 const auto& [voxel, voxel_points] = *it;
473 const auto& pt = voxel_points.front();
474 if ((pt.pos - origin).squaredNorm() >= max_distance2) {
475 it = this->get_active_map().erase(it);
476 counter++;
477 } else {
478 ++it;
479 }
480 }
481 // Update debug information
482 this->debug_.voxel_removed = counter;
483 this->debug_.num_voxel = this->get_active_map().size();
484 this->debug_.points_removed = this->num_points() - this->debug_.num_points;
485 this->debug_.num_points = this->num_points();
486 }
