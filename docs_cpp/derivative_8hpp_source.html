64 types::TangentStamped get_tangent([[maybe_unused]] const std::vector<types::Point<TConfig>>& frame,
65 const types::PoseStamped& pose_registered) override
66 {
67 const auto t_start = std::chrono::high_resolution_clock::now();
68 types::TangentStamped result{};
69 result.stamp = pose_registered.stamp;
70
71 // Return if the buffer is empty or the handler is not initialized
72 if (!previous_pose_.has_value()) {
73 std::cerr << "[VelocityHandler::Derivative]: Velocity estimation not initialized!" << std::endl;
74 previous_pose_ = pose_registered;
75 result.tangent << Eigen::Matrix<float, 6, 1>::Zero();
76 finalize(t_start, 0.0);
77 return result;
78 }
79
80 // Compute difference in timestamps to the last buffered pose
81 const double dt = (pose_registered.stamp - previous_pose_.value().stamp) * 1e-9;
82
83 // Check for invalid time delta
84 if (!std::isfinite(dt) || std::abs(dt) < 1e-6) {
85 std::cerr << "[VelocityHandler::Derivative]: Invalid time delta: " << dt << std::endl;
86 // Keep previous velocity or set zero
87 previous_pose_ = pose_registered;
88 result.tangent << Eigen::Matrix<float, 6, 1>::Zero();
89 finalize(t_start, dt);
90 return result;
91 }
92 // Check if dt is too large
93 if (std::abs(dt) > 1.0f) {
94 std::cerr << "[VelocityHandler::Derivative]: Time delta too large: " << dt << "s. Resetting velocity."
95 << std::endl;
96 previous_pose_ = pose_registered;
97 result.tangent << Eigen::Matrix<float, 6, 1>::Zero();
98 finalize(t_start, dt);
99 return result;
100 }
101
102 // Compute velocity using the logarithmic map of the SE3 difference
103 // ==== Extract of the sophus docs ====
104 // For the derivation of the logarithm of SE(3), see
105 // J. Gallier, D. Xu, "Computing exponentials of skew symmetric matrices
106 // and logarithms of orthogonal matrices", IJRA 2002.
107 // https:///pdfs.semanticscholar.org/cfe3/e4b39de63c8cabd89bf3feff7f5449fc981d.pdf
108 // (Sec. 6., pp. 8)
109 // ====================================
110 const Sophus::SE3f delta = previous_pose_.value().pose.inverse() * pose_registered.pose;
111 result.tangent = delta.log() / static_cast<float>(dt);
112
113 // Validate linear velocity (check for NaNs and unreasonably high speeds)
114 if (!result.tangent.head<3>().allFinite() || result.tangent.head<3>().norm() > 100.0f) {
115 std::cerr << "[VelocityHandler::Derivative]: Invalid linear velocity computed: "
116 << result.tangent.head<3>().transpose() << std::endl;
117 result.tangent.head<3>() = Eigen::Vector3f::Zero();
118 }
119 // Validate angular velocity (check for NaNs and unreasonably high speeds)
120 if (!result.tangent.tail<3>().allFinite() || result.tangent.tail<3>().norm() > 10.0f) {
121 std::cerr << "[VelocityHandler::Derivative]: Invalid angular velocity computed: "
122 << result.tangent.tail<3>().transpose() << std::endl;
123 result.tangent.tail<3>() = Eigen::Vector3f::Zero();
124 }
125 // Buffer the current pose for next iteration
126 previous_pose_ = pose_registered;
127 finalize(t_start, dt);
128 return result;
129 }

149 }
