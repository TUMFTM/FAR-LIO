55 void set_pose(const types::PoseStamped& pose, const bool valid) override
56 {
57 this->previous_guess_ = this->current_guess_;
58 // Valid pose, just set it
59 if (valid) {
60 this->current_guess_ = pose;
61 } else {
62 std::cerr << "[ModelHandler]: Invalid pose received, using CV prediction!" << std::endl;
63 if (this->current_guess_.has_value()) {
64 // Compute new pose predicting the difference between to the last pose forward
65 Sophus::SE3f predicted_pose = utils::cv_prediction(pose.pose, this->current_guess_.value().pose);
66 // Update current pose
67 this->current_guess_.value().pose = predicted_pose;
68 this->current_guess_.value().stamp = pose.stamp;
69 } else {
70 std::cerr << "[ModelHandler]: Model not initialized, initializing to config!" << std::endl;
71 this->init_model_config();
72 }
73 }
74 };

79 types::PoseStamped get_initial_guess([[maybe_unused]] std::uint64_t stamp) override
80 {
81 if (!this->current_guess_.has_value()) {
82 std::cerr << "[ModelHandler]: Prediction model not initialized! - Return empty pose!" << std::endl;
83 return types::PoseStamped{};
84 }
85 this->debug_.current_pos_x = this->current_guess_.value().pose.translation().x();
86 this->debug_.current_pos_y = this->current_guess_.value().pose.translation().y();
87 this->debug_.current_pos_z = this->current_guess_.value().pose.translation().z();
88 this->debug_.current_rot_w = this->current_guess_.value().pose.so3().unit_quaternion().w();
89 this->debug_.current_rot_x = this->current_guess_.value().pose.so3().unit_quaternion().x();
90 this->debug_.current_rot_y = this->current_guess_.value().pose.so3().unit_quaternion().y();
91 this->debug_.current_rot_z = this->current_guess_.value().pose.so3().unit_quaternion().z();
92 return this->current_guess_.value();
93 };
