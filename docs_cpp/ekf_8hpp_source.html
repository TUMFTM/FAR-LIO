57 void update(const Eigen::Ref<const Eigen::Vector<double, TConfig::MEASUREMENT_VECTOR_SIZE>>& z)
58 {
59 // calculate the innovation/ measurement residuals
60 residuals_raw_ = z - H_ * x_pred_;
61
62 // normalize all residual angles
63 for (int i = 0; i < TConfig::NUM_ORIENTATION_MEASUREMENT; ++i) {
64 for (int j = 0; j < TConfig::ORIENTATION_MEASUREMENT_VECTOR_SIZE; j++) {
65 const uint8_t vec_offset =
66 TConfig::MEASUREMENT_VECTOR_OFFSET_ORIENTATION + i * TConfig::ORIENTATION_MEASUREMENT_VECTOR_SIZE + j;
67 residuals_raw_[vec_offset] = tam::helpers::geometry::normalize_angle(residuals_raw_[vec_offset]);
68 }
69 }
70
71 // Calculate the innovation (residual) covariance S
72 const Eigen::Matrix<double, TConfig::MEASUREMENT_VECTOR_SIZE, TConfig::MEASUREMENT_VECTOR_SIZE> S =
73 H_ * P_pred_ * H_.transpose() + R_;
74
75 // Calculate the Kalman Gain K
76 K_.noalias() = P_pred_ * H_.transpose() * S.inverse();
77
78 // perform the outlier filtering
79 // the maximum allowed residum is defined in the variable P_VDC_OutlierBounds
80 residuals_raw_ = residuals_raw_.cwiseProduct(fusion_vec_);
81 if (kalman_filter_params_.enable_mahalanobis_outlier_rejection) {
82 residuals_ = outlier_rejection::mahalanobis_outlier_rejection<TConfig>(residuals_raw_, outlier_bound_, x_);
83 } else {
84 residuals_ = outlier_rejection::box_outlier_rejection(residuals_raw_, outlier_bound_);
85 }
86
87 // set the residual to zero if the sensor was not updated or is not valid
88 residuals_ = residuals_.cwiseProduct(fusion_vec_);
89
90 // update the state vector w.r.t the Kalman Gain
91 x_ = x_pred_ + K_ * residuals_;
92
93 // normalize all angles of the state vector
94 for (int i = 0; i < TConfig::STATE_VECTOR_SIZE; i++) {
95 if (TConfig::STATE_VECTOR_ANGLE_INDICATOR[i]) {
96 x_[i] = tam::helpers::geometry::normalize_angle(x_[i]);
97 }
98 }
99
100 // update the state covariance matrix
101 P_.noalias() = (Eigen::Matrix<double, TConfig::STATE_VECTOR_SIZE, TConfig::STATE_VECTOR_SIZE>::Identity() - K_ * H_) * P_pred_;
102 }
