20template <>
21EKF<tam::core::state::EKF_2D>::EKF()
22{
23 using TConfig = tam::core::state::EKF_2D;
24 // ensure that all matrices are set to zero
25 A_.setZero();
26 B_.setZero();
27
28 // Initialize EKF in 2D
29 // set the H_full matrix (this matrix would fuse all measurements)
30 // first set the rows that correspond to the localization measurements
31 // clang-format off
32 for (int i = 0; i < TConfig::NUM_POS_MEASUREMENT; ++i) {
33 H_full_.row(i * TConfig::POS_MEASUREMENT_VECTOR_SIZE + TConfig::MEASUREMENT_POS_X_M) << 1, 0, 0, 0, 0; // NOLINT
34 H_full_.row(i * TConfig::POS_MEASUREMENT_VECTOR_SIZE + TConfig::MEASUREMENT_POS_Y_M) << 0, 1, 0, 0, 0; // NOLINT
35 }
36 // clang-format on
37
38 // in the next step we set the rows that correspond to the orientation measurements
39 // clang-format off
40 for (int i = 0; i < TConfig::NUM_ORIENTATION_MEASUREMENT; ++i) {
41 H_full_.row(i * TConfig::ORIENTATION_MEASUREMENT_VECTOR_SIZE + TConfig::MEASUREMENT_VECTOR_OFFSET_ORIENTATION + TConfig::MEASUREMENT_PSI_RAD) << 0, 0, 1, 0, 0; // NOLINT
42 }
43 // clang-format on
44
45 // in the next step we set the rows that correspond to the velocity measurements
46 // clang-format off
47 for (int i = 0; i < TConfig::NUM_VEL_MEASUREMENT; ++i) {
48 H_full_.row(i * TConfig::VEL_MEASUREMENT_VECTOR_SIZE + TConfig::MEASUREMENT_VECTOR_OFFSET_VEL + TConfig::MEASUREMENT_VX_MPS) << 0, 0, 0, 1, 0; // NOLINT
49 H_full_.row(i * TConfig::VEL_MEASUREMENT_VECTOR_SIZE + TConfig::MEASUREMENT_VECTOR_OFFSET_VEL + TConfig::MEASUREMENT_VY_MPS) << 0, 0, 0, 0, 1; // NOLINT
50 }
51 // clang-format on
52
53 this->declare_and_update_parameters();
54}

62template <>
63void EKF<tam::core::state::EKF_2D>::predict(
64 const Eigen::Ref<const Eigen::Vector<double, tam::core::state::EKF_2D::INPUT_VECTOR_SIZE>>& u)
65{
66 using TConfig = tam::core::state::EKF_2D;
67 if (param_manager_->get_state_hash() != previous_param_state_hash_) {
68 this->declare_and_update_parameters();
69 }
70
71 // Prediction Step of the 2D EKF
72 // define the following functions for better readability
73 // previous state vector
74 const double psi_rad = x_[TConfig::STATE_PSI_RAD];
75 const double vx_mps = x_[TConfig::STATE_VX_MPS];
76 const double vy_mps = x_[TConfig::STATE_VY_MPS];
77
78 // new input vector
79 const double dPsi_radps = u[TConfig::INPUT_DPSI_RADPS];
80 const double ax_mps2 = u[TConfig::INPUT_AX_MPS2];
81 const double ay_mps2 = u[TConfig::INPUT_AY_MPS2];
82
83 // construct the Jacobian matrix consisting of partial derivatives with respect to the
84 // system state from the system update equation
85 // clang-format off
86 A_ << 0, 0, (-std::sin(psi_rad) * vx_mps - std::cos(psi_rad) * vy_mps), std::cos(psi_rad), -std::sin(psi_rad), // NOLINT
87 0, 0, (std::cos(psi_rad) * vx_mps - std::sin(psi_rad) * vy_mps), std::sin(psi_rad), std::cos(psi_rad), // NOLINT
88 0, 0, 0, 0, 0, // NOLINT
89 0, 0, 0, 0, dPsi_radps, // NOLINT
90 0, 0, 0, -dPsi_radps, 0; // NOLINT
91 // clang-format on
92
93 A_ = Eigen::Matrix<double, TConfig::STATE_VECTOR_SIZE, TConfig::STATE_VECTOR_SIZE>::Identity() + A_ * TConfig::TS;
94
95 // Jacobian matrix B consisting of partial derivatives with respect to the
96 // system input from the system update equation
97 // clang-format off
98 B_ << 0, 0, 0, // NOLINT
99 0, 0, 0, // NOLINT
100 1, 0, 0, // NOLINT
101 vy_mps, 1, 0, // NOLINT
102 -vx_mps, 0, 1; // NOLINT
103 // clang-format on
104
105 B_ = TConfig::TS * B_;
106
107 // fuse the process noise covariance matrix with
108 // the linearized version of the system input vector (non-additive noise)
109 const Eigen::Matrix<double, TConfig::STATE_VECTOR_SIZE, TConfig::STATE_VECTOR_SIZE> Q_lin = B_ * Q_ * B_.transpose();
110
111 // State Transition:
112 // Rigid body model using accelerations and yaw rate as inputs and applying simple forward
113 // integration in vehicle coordinate frame to obtain the velocities. The latter are then
114 // used to apply forward integration in global cartesian coordinates to obtain the
115 // position. Using the ENU convention (0 degrees heading is east (x-axis)).
116 Eigen::Vector<double, TConfig::STATE_VECTOR_SIZE> dx;
117 // clang-format off
118 dx << std::cos(psi_rad) * vx_mps - std::sin(psi_rad) * vy_mps, // NOLINT
119 std::sin(psi_rad) * vx_mps + std::cos(psi_rad) * vy_mps, // NOLINT
120 dPsi_radps, // NOLINT
121 ax_mps2 + dPsi_radps * vy_mps, // NOLINT
122 ay_mps2 - dPsi_radps * vx_mps; // NOLINT
123 // clang-format on
124
125 // integrate over on timestep
126 x_pred_ = x_ + TConfig::TS * dx;
127
128 // Normalize the angles of every angle in the state transition vector
129 for (int i = 0; i < TConfig::STATE_VECTOR_SIZE; i++) {
130 if (TConfig::STATE_VECTOR_ANGLE_INDICATOR[i] == true) {
131 x_pred_[i] = tam::helpers::geometry::normalize_angle(x_pred_[i]);
132 }
133 }
134
135 // Predict the state covariance
136 P_pred_.noalias() = A_ * P_ * A_.transpose() + Q_lin;
137}

145template <>
146const tam::types::state::unordered_identifier_map<std::vector<std::pair<std::string, double>>>&
147EKF<tam::core::state::EKF_2D>::get_debug(void)
148{
149 using TConfig = tam::core::state::EKF_2D;
150 // get the debug outputs of the 2D EKF
151 // Iterate through all position residuals
152 for (int i = 0; i < TConfig::NUM_POS_MEASUREMENT; ++i) {
153 // Get a reference to the vector for this ID once
154 auto& debug = debug_output_[{tam::types::state::measurements::POS, static_cast<uint8_t>(i)}];
155 debug.clear();
156
157 // cartesian position residuals in meters
158 const uint8_t idx_pos_x = i * TConfig::POS_MEASUREMENT_VECTOR_SIZE + TConfig::MEASUREMENT_POS_X_M;
159 const uint8_t idx_pos_y = i * TConfig::POS_MEASUREMENT_VECTOR_SIZE + TConfig::MEASUREMENT_POS_Y_M;
160 debug.emplace_back("res_x_m", residuals_raw_[idx_pos_x]);
161 debug.emplace_back("res_y_m", residuals_raw_[idx_pos_y]);
162
163 // cartesian position measurement variance
164 debug.emplace_back("R_x_m", R_(idx_pos_x, idx_pos_x));
165 debug.emplace_back("R_y_m", R_(idx_pos_y, idx_pos_y));
166
167 // frenet position residuals in meters
168 // clang-format off
169 debug.emplace_back("res_s_m", std::cos(x_[TConfig::STATE_PSI_RAD]) * residuals_raw_[idx_pos_x] + std::sin(x_[TConfig::STATE_PSI_RAD]) * residuals_raw_[idx_pos_y]); // NOLINT
170 debug.emplace_back("res_d_m", -std::sin(x_[TConfig::STATE_PSI_RAD]) * residuals_raw_[idx_pos_x] + std::cos(x_[TConfig::STATE_PSI_RAD]) * residuals_raw_[idx_pos_y]); // NOLINT
171 // clang-format on
172 }
173
174 // Iterate through all orientation residuals
175 for (int i = 0; i < TConfig::NUM_ORIENTATION_MEASUREMENT; ++i) {
176 // Get a reference to the vector for this ID once
177 auto& debug = debug_output_[{tam::types::state::measurements::ORIENTATION, static_cast<uint8_t>(i)}];
178 debug.clear();
179
180 // cartesian orientation residual and measurement in rad
181 // clang-format off
182 const uint8_t idx_psi = TConfig::MEASUREMENT_VECTOR_OFFSET_ORIENTATION + i * TConfig::ORIENTATION_MEASUREMENT_VECTOR_SIZE + TConfig::MEASUREMENT_PSI_RAD; // NOLINT
183 debug.emplace_back("res_psi_rad", residuals_raw_[idx_psi]);
184 debug.emplace_back("R_psi", R_(idx_psi, idx_psi));
185 // clang-format on
186 }
187
188 // Iterate through all linear velocity residuals
189 for (int i = 0; i < TConfig::NUM_VEL_MEASUREMENT; ++i) {
190 // Get a reference to the vector for this ID once
191 auto& debug = debug_output_[{tam::types::state::measurements::VEL, static_cast<uint8_t>(i)}];
192 debug.clear();
193
194 // velocity residuals in the vehicle frame in meters
195 // clang-format off
196 const uint8_t idx_pos_vx = TConfig::MEASUREMENT_VECTOR_OFFSET_VEL + i * TConfig::VEL_MEASUREMENT_VECTOR_SIZE + TConfig::MEASUREMENT_VX_MPS; // NOLINT
197 const uint8_t idx_pos_vy = TConfig::MEASUREMENT_VECTOR_OFFSET_VEL + i * TConfig::VEL_MEASUREMENT_VECTOR_SIZE + TConfig::MEASUREMENT_VY_MPS; // NOLINT
198 debug.emplace_back("res_vx_mps", residuals_raw_[idx_pos_vx]);
199 debug.emplace_back("res_vy_mps", residuals_raw_[idx_pos_vy]);
200 // clang-format on
201 }
202 return debug_output_;
203}
