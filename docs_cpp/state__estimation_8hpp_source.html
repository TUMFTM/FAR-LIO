108 void step(void) override
109 {
110 if (param_manager_->get_state_hash() != previous_param_state_hash_) {
111 this->declare_and_update_parameters();
112 }
113 vehicle_model_handler_->declare_and_update_parameters();
114
115 // preprocess the input vector containing linear accelerations and angular veocities
116 // the valid_u_fusion_vec ensures that only valid measurements are fused
117 u_ = imu_handler_->update_input_vector(u_raw_, state_machine_->get_valid_u_fusion_vec(), u_cov_);
118 // update the reference orientation based on the updated imu measurements
119 // and the last predicted vehicle odometry
120 if (state_estimation_params_.fuse_ref_angles) {
121 // set the output as last orientation measurement
122 const tam::types::control::Odometry ref_orientation = ref_orientation_handler_->update(x_out_, u_);
123 ref_angles_ = ref_orientation.orientation_rad;
124 StateEstimation<TConfig, TModel>::set_input_orientation(
125 ref_orientation, TConfig::NUM_ORIENTATION_MEASUREMENT - 1, true, true, false);
126 StateEstimation<TConfig, TModel>::set_input_orientation_status(
127 ref_orientation_handler_->get_status(), TConfig::NUM_ORIENTATION_MEASUREMENT - 1);
128 }
129
130 // update the measurement matrix of the Kalman filter to only fuse updated and valid measurements
131 if (kalman_filter_->get_enable_adaptive_measurement_covariance()) {
132 // update the measurement covariance matrix of the Kalman filter
133 kalman_filter_->update_measurement_covariance_matrix(state_machine_->get_debug());
134 }
135 kalman_filter_->update_measurement_matrix(fusion_vec_.cwiseProduct(state_machine_->get_valid_fusion_vec()));
136
137 // buffer the measurement vector to ensure that no asynchronous measurements
138 // are received during the prediction step
139 const Eigen::Vector<double, TConfig::MEASUREMENT_VECTOR_SIZE> z_sync = z_;
140
141 // set the fusion vector to zero to only consider fresh measurements in the next step
142 fusion_vec_.setZero();
143
144 // perform the prediction step of the Kalman filter
145 kalman_filter_->predict(u_);
146
147 // perform the update step of the Kalman filter
148 kalman_filter_->update(z_sync);
149
150 // get the Kalman state vector
151 x_out_ = kalman_filter_->get_state_vector();
152
153 // overwrite the state machine if the state estimation is in a safe state
154 const auto& covariance = kalman_filter_->get_covariance_matrix();
155 const double pos_x_variance = covariance(TConfig::STATE_POS_X_M, TConfig::STATE_POS_X_M);
156 const double pos_y_variance = covariance(TConfig::STATE_POS_Y_M, TConfig::STATE_POS_Y_M);
157 const double mean_position_variance = (pos_x_variance + pos_y_variance) / 2;
158 if (mean_position_variance < state_estimation_params_.overwrite_pose_covariance_threshold) {
159 state_machine_->update(true);
160 } else {
161 state_machine_->update();
162 }
163
164 // set the imu offsets
165 StateEstimation<TConfig, TModel>::update_imu_offsets();
166 }

174 bool set_initial_state(void) override
175 {
176 if (param_manager_->get_state_hash() != previous_param_state_hash_) {
177 this->declare_and_update_parameters();
178 }
179
180 // set the sensor biases for the imu handler (robust to a bias_init that is shorter/longer than the input vector)
181 Eigen::Vector<double, TConfig::INPUT_VECTOR_SIZE> imu_bias =
182 Eigen::Vector<double, TConfig::INPUT_VECTOR_SIZE>::Zero();
183 const int num_bias =
184 std::min<int>(static_cast<int>(state_estimation_params_.u_bias_init.size()), TConfig::INPUT_VECTOR_SIZE);
185 for (int i = 0; i < num_bias; ++i) imu_bias[i] = state_estimation_params_.u_bias_init[i];
186 imu_handler_->set_sensor_bias(imu_bias);
187
188 // state vector to overwrite the original one
189 Eigen::Vector<double, TConfig::STATE_VECTOR_SIZE> x_overwrite = Eigen::VectorXd::Zero(TConfig::STATE_VECTOR_SIZE);
190 bool input_valid = false;
191 state_machine_->update(true);
192
193 // Lambda to set the inital state of the kalman filter given a initial x vector
194 auto set_initial_kalman_state =
195 [this](const Eigen::Ref<const Eigen::Vector<double, TConfig::STATE_VECTOR_SIZE>> x_overwrite) {
196 // set the initial state of the kalman filter
197 this->kalman_filter_->set_state_vector(x_overwrite);
198 this->kalman_filter_->set_covariance_matricies();
199 this->kalman_filter_->set_initial_valid_map(state_machine_->get_debug());
200 this->x_out_ = x_overwrite;
201
202 // set the initial state of the State Machine
203 // overwrite_state_machine = true allows all state transitions
204 this->state_machine_->update(true);
205 };
206
207 // ensure that enough imus are valid to not trigger a safe stop after init
208 if (state_machine_->get_min_valid_imus()) return false;
209
210 // initialize the kalman filter with zeros
211 if (!state_estimation_params_.initialize_from_measurements) {
212 set_initial_kalman_state(x_overwrite);
213 return true;
214 }
215
216 // to ensure that the initial position is valid the state vector of the kalman filter
217 // is only overwritten iff the measurement is valid and updated
218 fusion_vec_ = fusion_vec_.cwiseProduct(state_machine_->get_valid_fusion_vec());
219 for (int i = 0; i < TConfig::NUM_POS_MEASUREMENT; ++i) {
220 // clang-format off
221 const Eigen::Vector<double, TConfig::POS_MEASUREMENT_VECTOR_SIZE> position_elements = fusion_vec_.segment(i * TConfig::POS_MEASUREMENT_VECTOR_SIZE, TConfig::POS_MEASUREMENT_VECTOR_SIZE); // NOLINT
222 if (position_elements.isApprox(Eigen::VectorXd::Ones(TConfig::POS_MEASUREMENT_VECTOR_SIZE))) {
223 x_overwrite.segment(0, TConfig::POS_MEASUREMENT_VECTOR_SIZE) = z_.segment(i * TConfig::POS_MEASUREMENT_VECTOR_SIZE, TConfig::POS_MEASUREMENT_VECTOR_SIZE); // NOLINT
224 input_valid = true;
225 break;
226 }
227 // clang-format on
228 }
229
230 const Eigen::Vector<double, 2> orientation(external_orientation_.x, external_orientation_.y);
231 // clang-format off
232 const int orientation_vector_offset = TConfig::MEASUREMENT_VECTOR_OFFSET_ORIENTATION + (TConfig::NUM_ORIENTATION_MEASUREMENT - 2) * TConfig::ORIENTATION_MEASUREMENT_VECTOR_SIZE; // NOLINT
233 // clang-format on
234 // set the orientation in the measurement vector (penultimate measurement)
235 z_.segment(orientation_vector_offset, 2) = orientation;
236
237 // set bits in the fusion vector to indicate that a new measurement was received
238 fusion_vec_.segment(orientation_vector_offset, 2) = Eigen::VectorXd::Ones(2);
239
240 // set the initial orientation of the state vector iff a valid position was found
241 // we have to search for every axis individually because not every
242 // orientation measurement contains all axis
243 if (input_valid) {
244 for (int i = 0; i < TConfig::ORIENTATION_MEASUREMENT_VECTOR_SIZE; ++i) {
245 input_valid = false;
246 for (int j = 0; j < TConfig::NUM_ORIENTATION_MEASUREMENT; j++) {
247 // clang-format off
248 const int idx = TConfig::MEASUREMENT_VECTOR_OFFSET_ORIENTATION + j * TConfig::ORIENTATION_MEASUREMENT_VECTOR_SIZE + i; // NOLINT
249 // clang-format on
250 if (fusion_vec_[idx] > 0.0) {
251 x_overwrite[TConfig::POS_MEASUREMENT_VECTOR_SIZE + i] = z_[idx];
252 input_valid = true;
253 break;
254 }
255 }
256 if (!input_valid) break;
257 }
258 }
259
260 // if a valid position and orientation was found also search for a valid velocity measurement
261 if (input_valid) {
262 for (int i = 0; i < TConfig::NUM_VEL_MEASUREMENT; ++i) {
263 const int idx = TConfig::MEASUREMENT_VECTOR_OFFSET_VEL + i * TConfig::VEL_MEASUREMENT_VECTOR_SIZE;
264 if (fusion_vec_[idx] > 0.0) {
265 // clang-format off
266 x_overwrite.segment(TConfig::POS_MEASUREMENT_VECTOR_SIZE + TConfig::ORIENTATION_MEASUREMENT_VECTOR_SIZE, TConfig::VEL_MEASUREMENT_VECTOR_SIZE) = z_.segment(idx, TConfig::VEL_MEASUREMENT_VECTOR_SIZE); // NOLINT
267 // clang-format on
268 // set the initial state of the kalman filter
269 set_initial_kalman_state(x_overwrite);
270 return true;
271 }
272 }
273 return false;
274 }
275 return false;
276 }

290 // position input
299 void set_input_position(const tam::types::control::Odometry& odometry, uint8_t pos_num) override
300 {
301 // return if the input contains a NaN
302 if (odometry.position_m.hasNaN()) {
303 state_machine_->set_position_invalid(pos_num);
304 return;
305 }
306
307 if constexpr (!TConfig::THREE_DIMENSIONAL) {
308 // sets the positional input of 2D filters
309 // set the measurment vector
310 if (pos_num < TConfig::NUM_POS_MEASUREMENT) {
311 const int pos_measurement_idx = pos_num * TConfig::POS_MEASUREMENT_VECTOR_SIZE;
312 // set position measurement vector
313 z_[pos_measurement_idx + TConfig::MEASUREMENT_POS_X_M] = odometry.position_m.x;
314 z_[pos_measurement_idx + TConfig::MEASUREMENT_POS_Y_M] = odometry.position_m.y;
315
316 // set the bits corresponding to this position input in the fusion_vec to one to fuse them
317 // iff the position is also valid in the state machine
318 // clang-format off
319 fusion_vec_.segment(pos_measurement_idx, TConfig::POS_MEASUREMENT_VECTOR_SIZE) = Eigen::VectorXd::Ones(TConfig::POS_MEASUREMENT_VECTOR_SIZE); // NOLINT
320 // clang-format on
321
322 // update the variance of the sensor to adapt the R matrix of the kalman filter
323 if (kalman_filter_->get_enable_adaptive_measurement_covariance()) {
324 Eigen::Vector<double, TConfig::POS_MEASUREMENT_VECTOR_SIZE> covariance;
325 covariance << odometry.pose_covariance[0], odometry.pose_covariance[7];
326 kalman_filter_->set_position_covariance(covariance, pos_num);
327 }
328 } else {
329 throw std::invalid_argument(
330 "[StateEstimationCPP]: Localization input " + std::to_string(pos_num) + " does not exist");
331 }
332 } else {
333 // sets the positional input 3D filters
334 if (pos_num < TConfig::NUM_POS_MEASUREMENT) {
335 const int pos_measurement_idx = pos_num * TConfig::POS_MEASUREMENT_VECTOR_SIZE;
336 // set position measurement vector
337 z_[pos_measurement_idx + TConfig::MEASUREMENT_POS_X_M] = odometry.position_m.x;
338 z_[pos_measurement_idx + TConfig::MEASUREMENT_POS_Y_M] = odometry.position_m.y;
339 z_[pos_measurement_idx + TConfig::MEASUREMENT_POS_Z_M] = odometry.position_m.z;
340
341 // set the bits corresponding to this position input in the fusion_vec to one to fuse them
342 // iff the position is also valid in the state machine
343 // clang-format off
344 fusion_vec_.segment(pos_measurement_idx, TConfig::POS_MEASUREMENT_VECTOR_SIZE) = Eigen::VectorXd::Ones(TConfig::POS_MEASUREMENT_VECTOR_SIZE); // NOLINT
345 // clang-format on
346
347 // update the variance of the sensor to adapt the R matrix of the kalman filter
348 if (kalman_filter_->get_enable_adaptive_measurement_covariance()) {
349 Eigen::Vector<double, TConfig::POS_MEASUREMENT_VECTOR_SIZE> covariance;
350 covariance << odometry.pose_covariance[0], odometry.pose_covariance[7], odometry.pose_covariance[14];
351 kalman_filter_->set_position_covariance(covariance, pos_num);
352 }
353 } else {
354 throw std::invalid_argument(
355 "[StateEstimationCPP]: Position input " + std::to_string(pos_num) + " does not exist");
356 }
357 }

551 // linear velocity inputs
560 void set_input_linear_velocity(const tam::types::control::Odometry& odometry, uint8_t vel_num) override
561 {
562 // return if the input contains a NaN
563 if (odometry.velocity_mps.hasNaN()) {
564 state_machine_->set_linear_velocity_invalid(vel_num);
565 return;
566 }
567
568 if constexpr (!TConfig::THREE_DIMENSIONAL) {
569 // sets the linear velocity input in 2D
570 // set the measurment vector
571 if (vel_num < TConfig::NUM_VEL_MEASUREMENT) {
572 // clang-format off
573 const int vel_measurement_idx = TConfig::MEASUREMENT_VECTOR_OFFSET_VEL + vel_num * TConfig::VEL_MEASUREMENT_VECTOR_SIZE; // NOLINT
574 // transform the 3D linear velocities into the footprint frame
575 tam::types::common::Vector3D<double> odometry_footprint;
576 odometry_footprint = tam::helpers::euler_rotations::vector_to_2d(odometry.velocity_mps, external_orientation_.y, external_orientation_.x); // NOLINT
577 // set the linear velocity measurements
578 z_[vel_measurement_idx + TConfig::MEASUREMENT_VX_MPS] = odometry_footprint.x;
579 z_[vel_measurement_idx + TConfig::MEASUREMENT_VY_MPS] = odometry_footprint.y;
580 // set the bits corresponding to this linear velocity input in the fusion_vec to one
581 // to fuse them iff the localization is also valid in the state machine
582 fusion_vec_.segment(vel_measurement_idx, TConfig::VEL_MEASUREMENT_VECTOR_SIZE) = Eigen::VectorXd::Ones(TConfig::VEL_MEASUREMENT_VECTOR_SIZE); // NOLINT
583 // clang-format on
584 } else {
585 throw std::invalid_argument(
586 "[StateEstimationCPP]: Linear velocity input " + std::to_string(vel_num) + " does not exist");
587 }
588 } else {
589 // set the linear velocity input in 3D
590 // set the measurment vector
591 if (vel_num < TConfig::NUM_VEL_MEASUREMENT) {
592 // clang-format off
593 const int vel_measurement_idx = TConfig::MEASUREMENT_VECTOR_OFFSET_VEL + vel_num * TConfig::VEL_MEASUREMENT_VECTOR_SIZE; // NOLINT
594 const double outlier_threshold = state_estimation_params_.hard_linear_velocity_outlier_th;
595 // set the linear velocity measurements
596 z_[vel_measurement_idx + TConfig::MEASUREMENT_VX_MPS] = odometry.velocity_mps.x;
597 z_[vel_measurement_idx + TConfig::MEASUREMENT_VY_MPS] = odometry.velocity_mps.y;
598 z_[vel_measurement_idx + TConfig::MEASUREMENT_VZ_MPS] = odometry.velocity_mps.z;
599 // completly reject measurements that violate the hard treshold
600 const Eigen::Vector<double, TConfig::VEL_MEASUREMENT_VECTOR_SIZE> outlier_distance_se = tam::core::state::outlier_rejection::squared_distance(x_out_.segment(TConfig::STATE_VX_MPS, TConfig::VEL_MEASUREMENT_VECTOR_SIZE), odometry.velocity_mps); // NOLINT
601
602 if ((outlier_distance_se.array() < outlier_threshold).all() || (0.0 == x_out_.segment(TConfig::STATE_VX_MPS, TConfig::VEL_MEASUREMENT_VECTOR_SIZE).array()).all()) { // NOLINT

1078 // clang-format on

1080 // x,y,z coordintates
1081 output.position_m.x = x_out_[TConfig::STATE_POS_X_M];
1082 output.position_m.y = x_out_[TConfig::STATE_POS_Y_M];
1083

1086 output.orientation_rad.y = 0.0;

1094 output.angular_velocity_radps.z = u_[TConfig::INPUT_DPSI_RADPS];
1095
1096 // set the covariance matrix for the position
1097 output.pose_covariance[0] = P(0, 0);
1098 output.pose_covariance[1] = P(0, 1);
1099 output.pose_covariance[6] = P(1, 0);
1100 output.pose_covariance[7] = P(1, 1);
1101
1102 // set the covariance matrix for the orientation
1103 output.pose_covariance[35] = P(2, 2);
1104
1105 // set the covariance matrix for the linear velocity
1106 output.velocity_covariance[0] = P(3, 3);
1107 output.velocity_covariance[1] = P(3, 4);
1108 output.velocity_covariance[6] = P(4, 3);

1115 // clang-format on

1118 output.position_m.x = x_out_[TConfig::STATE_POS_X_M];

1126

1142 output.angular_velocity_radps.z = u_[TConfig::INPUT_DPSI_RADPS];

1145 for (int i = 6; i < 9; ++i) {
