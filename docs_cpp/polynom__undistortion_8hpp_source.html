56 void undistort(std::vector<types::Point<TConfig>>& frame, const std::uint64_t frame_stamp) override
57 {
58 // Return early if no pose history is available or the velocity is below the threshold
59 // clang-format off
60 if (this->pose_history_.size() < 2 || this->estimate_velocity() < this->config_.vel_threshold) { // NOLINT
61 this->debug_.valid = false;
62 this->debug_.undistortion_time = 0.0;
63 this->debug_.timestamp_offset = 0.0;
64 this->debug_.num_invalid_points = frame.size();
65 return;
66 }
67
68 auto start = std::chrono::high_resolution_clock::now();
69 Eigen::Matrix<float, 6, types::POLYNOM_DEGREE + 1> poly_coeff = Eigen::Matrix<float, 6, types::POLYNOM_DEGREE + 1>::Zero(); // NOLINT
70 float timestamp_offset = 0.0;
71 const bool valid = this->fit_polynom(frame_stamp, poly_coeff, timestamp_offset);
72 // clang-format on
73 // Fit polynomial to tangent history
74 if (!valid) {
75 this->debug_.valid = false;
76 this->debug_.undistortion_time = 0.0;
77 this->debug_.timestamp_offset = 0.0;
78 this->debug_.num_invalid_points = frame.size();
79 return;
80 }
81 int counter = 0;
82 // Iterate over frame using TBB parallel_for
83 std::atomic<int> atomic_counter{0};
84 using points_iterator = typename std::vector<types::Point<TConfig>>::iterator;
85 // clang-format off
86 const float min_timestamp = static_cast<float>((static_cast<double>(this->pose_history_.front().stamp) - static_cast<double>(frame_stamp)) * 1.0e-9); // NOLINT
87 const float max_timestamp = static_cast<float>((static_cast<double>(this->pose_history_.back().stamp) - static_cast<double>(frame_stamp)) * 1.0e-9); // NOLINT
88 const int degree = types::POLYNOM_DEGREE;
89 // clang-format on
90 tbb::parallel_for(tbb::blocked_range<points_iterator>(frame.begin(), frame.end()),
91 [&](const tbb::blocked_range<points_iterator>& r) {
92 std::for_each(r.begin(), r.end(), [&](auto& point) {
93 // Extract and clamp timestamp to avoid extrapolation.
94 float timestamp = std::clamp(point.timestamp, min_timestamp, max_timestamp);
95
96 // Evaluate polynomial using Horner's method -> log-pose relative to the reference frame
97 Sophus::SE3f::Tangent twist = Sophus::SE3f::Tangent();
98 const float t = timestamp + timestamp_offset;
99
100 for (int k = 0; k < 6; ++k) {
101 // Horner's method: evaluate from highest degree down
102 float sum = poly_coeff(k, degree);
103 for (int j = degree - 1; j >= 0; --j) {
104 sum = std::fma(sum, t, poly_coeff(k, j));
105 }
106 twist(k) = sum;
107 }
108
109 // Ensure that the correction is valid (no NaN / Inf)
110 if (twist.allFinite()) {
111 const auto motion = Sophus::SE3f::exp(twist);
112 point.pos = motion * point.pos;
113 } else {
114 atomic_counter++;
115 }
116 });
117 });
118 counter = atomic_counter.load();
119 // Compute elapsed time
120 // clang-format off
121 double unistortion_time = std::chrono::duration_cast<std::chrono::microseconds>(std::chrono::high_resolution_clock::now() - start).count() * 1.0e-3; // NOLINT
122 // clang-format on
123 // Update debug information
124 this->debug_.valid = valid;
125 this->debug_.undistortion_time = unistortion_time;
126 this->debug_.timestamp_offset = timestamp_offset;
127 this->debug_.num_invalid_points = static_cast<std::int64_t>(counter);
128 }
