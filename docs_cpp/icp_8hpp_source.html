38template <typename TConfig>
39struct ICPFactor : public FactorBase<TConfig> {
40 types::LinearSystem operator()(types::Correspondence<TConfig>& correspondence) const override
41 {
42 // Compute residual with double precision
43 const Eigen::Vector3f residual = correspondence.frame.pos - correspondence.map.pos;
44 Eigen::Matrix<float, 3, 6> J_r;
45 J_r.block<3, 3>(0, 0) = Eigen::Matrix3f::Identity();
46 J_r.block<3, 3>(0, 3) = -1.0 * Sophus::SO3f::hat(correspondence.frame.pos);
47 // Compute kernel weight
48 const float w = robust_kernel_weight<TConfig>(residual, this->kernel_scale_);
49 return types::LinearSystem(J_r.transpose() * w * J_r, // JTJ
50 J_r.transpose() * w * residual); // JTr
51 }
52};

62template <typename TConfig>
63struct ICPError : public ErrorBase<TConfig> {
64 float operator()(const types::Correspondence<TConfig>& correspondence) const override
65 {
66 // Compute residual with double precision
67 const Eigen::Vector3f residual = correspondence.map.pos - this->T_ * correspondence.frame.pos;
68 // Compute error
69 return 0.5f * residual.squaredNorm();
70 }
71};

106 Sophus::SE3f register_frame(const std::vector<types::Point<TConfig>>& frame, const MapHandler<TConfig>* map,
107 const Sophus::SE3f& initial_guess, const float correspondence_threshold, const float kernel_scale) override
108 {
109 if (map->empty()) return initial_guess;
110
111 // Prepare the frame
112 std::vector<types::Point<TConfig>> source = frame;
113 utils::transform_points(initial_guess, source);
114
115 // Call the solver
116 const Sophus::SE3f T =
117 this->solve(source, map, this->icp_factor_, this->icp_error_, correspondence_threshold, kernel_scale);
118
119 // Spit the final transformation
120 return T * initial_guess;
121 }
